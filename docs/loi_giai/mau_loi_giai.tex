\section{Câu 3. Mốc nội suy tối ưu}

\subsection{a) Xây dựng công thức $n$ mốc nội suy tối ưu trên $[-1,1]$ và trên $[a,b]$}

\pp{Sai số nội suy.} Với $n$ mốc $x_0,\dots,x_{n-1}\in[a,b]$ và $P$ bậc $\le n-1$:
\[ f(x)-P(x)=\frac{f^{(n)}(\xi)}{n!}\,w_n(x),\qquad w_n(x)=\prod_{i=0}^{n-1}(x-x_i). \]
$w_n$ là đa thức bậc $n$ có hệ số cao nhất bằng 1; mốc tối ưu là mốc làm $\max_{[a,b]}|w_n(x)|$ nhỏ nhất.

\pp{Trên $[-1,1]$.} Đa thức Chebyshev $T_n(t)=\cos(n\arccos t)$ có hệ số cao nhất $2^{n-1}$ và $|T_n(t)|\le1$. Trong các đa thức bậc $n$ có hệ số cao nhất bằng 1, $\tilde T_n=2^{1-n}T_n$ có $\max_{[-1,1]}|\tilde T_n|=2^{1-n}$ nhỏ nhất. Vậy chọn $w_n=\tilde T_n$, tức các mốc là nghiệm của $T_n$:
\[ \boxed{t_i=\cos\frac{(2i+1)\pi}{2n},\ i=0,\dots,n-1,\qquad \min\max_{[-1,1]}|w_n|=\frac{1}{2^{n-1}}.} \]

\pp{Trên $[a,b]$.} Đổi biến $x=\dfrac{a+b}{2}+\dfrac{b-a}{2}t$ ánh xạ $[-1,1]$ lên $[a,b]$. Khi đó $x-x_i=\frac{b-a}{2}(t-t_i)$ nên $w_n(x)=\left(\frac{b-a}{2}\right)^n\prod(t-t_i)$; bài toán trên $[a,b]$ tương đương bài toán trên $[-1,1]$:
\[ \boxed{x_i=\frac{a+b}{2}+\frac{b-a}{2}\cos\frac{(2i+1)\pi}{2n},\qquad \min\max_{[a,b]}|w_n|=\left(\frac{b-a}{2}\right)^n\frac{1}{2^{n-1}}=\frac{(b-a)^n}{2^{2n-1}},} \]
và khi đó $|f(x)-P(x)|\le\dfrac{M_n}{n!}\cdot\dfrac{(b-a)^n}{2^{2n-1}}$, $M_n=\max_{[a,b]}|f^{(n)}|$.

\pp{Sắp tăng dần.} $\cos$ giảm trên $[0,\pi]$ nên $t_i$ ở trên giảm theo $i$. Dùng $t_i=-\cos\frac{(2i+1)\pi}{2n}=\cos\frac{(2(n-1-i)+1)\pi}{2n}$ (cùng tập nghiệm, đánh số ngược) thì $t_0<t_1<\dots<t_{n-1}$.

\subsection{b) Thuật toán xác định $n$ mốc tối ưu sắp tăng dần trên $[a,b]$}
\begin{thuattoan}{Mốc nội suy tối ưu}
\item[Input:] số mốc $n\ge1$; đoạn $[a,b]$.
\item[Output:] các mốc $x_0<x_1<\dots<x_{n-1}$; $W=\max_{[a,b]}|w_n(x)|$.
\item[Bước 0:] Nếu $a=b$ thì báo lỗi, dừng thuật toán; nếu $a>b$ thì đổi chỗ $a$ và $b$.
\item[Bước 1:] Đặt $p:=\frac{a+b}{2}$, $q:=\frac{b-a}{2}$.
\item[Bước 2:] Lặp $i=0,\dots,n-1$: tính $t_i:=-\cos\dfrac{(2i+1)\pi}{2n}$ và $x_i:=p+q\,t_i$.
\item[Bước 3:] Tính $W:=\dfrac{(b-a)^n}{2^{2n-1}}$.\newline Dừng thuật toán, kết quả là $x=(x_0,\dots,x_{n-1})$ và $W$.
\end{thuattoan}

\section{Câu 4. Sơ đồ Horner: giá trị, phép chia, đạo hàm}

\subsection{a) Giá trị $P_n(c)$ \quad b) Chia $P_n(x)$ cho $(x-c)$}
\begin{thuattoan}{Horner chia (giá trị, thương, số dư)}
\item[Input:] $a=(a_n,\dots,a_0)$, $a_n\neq0$; số $c$.
\item[Output:] thương $b=(b_{n-1},\dots,b_0)$; số dư $r=P_n(c)$.
\item[Bước 0:] Bỏ các hệ số $0$ ở đầu vector $a$; nếu mọi hệ số bằng $0$ thì báo $P_n\equiv0$, dừng thuật toán.
\item[Bước 1:] Nếu $n=0$ thì đặt $r:=a_0$, dừng thuật toán.
\item[Bước 2:] Đặt $b_{n-1}:=a_n$.
\item[Bước 3:] Lặp $i=n-1,\dots,1$: tính $b_{i-1}:=a_i+c\,b_i$.
\item[Bước 4:] Tính $r:=a_0+c\,b_0$.\newline Dừng thuật toán, kết quả là $b$ và $r$ (chỉ cần giá trị thì lấy $r$).
\end{thuattoan}

\subsection{c) Đạo hàm cấp $k$ của $P_n$ tại $c$}
\begin{thuattoan}{Đạo hàm cấp $0..k$ tại $c$ (Horner chia lặp)}
\item[Input:] $a=(a_n,\dots,a_0)$; số $c$; cấp $k\ge0$.
\item[Output:] $r=(r_0,\dots,r_K)$, $K=\min(k,n)$; $v=(v_0,\dots,v_k)$ với $v_j=P_n^{(j)}(c)$.
\item[Bước 0:] Như Câu 4a.
\item[Bước 1:] Đặt $K:=\min(k,n)$ và $b_i:=a_i$, $i=0,\dots,n$.
\item[Bước 2:] Lặp $j=0,\dots,K$:
  \begin{con}
  \item[2.1:] Lặp $i=n-1,\dots,j$: tính $b_i:=b_i+c\,b_{i+1}$ (chia cho $x-c$).
  \item[2.2:] Đặt $r_j:=b_j$, tính $v_j:=j!\,r_j$.
  \end{con}
\item[Bước 3:] Lặp $j=K+1,\dots,k$: đặt $v_j:=0$.\newline Dừng thuật toán, kết quả là $r$ và $v$.
\end{thuattoan}

\section{Câu 6. Thuật toán đa thức nội suy Lagrange}

\pp{Công thức.} Với bộ điểm $(x_i,y_i)$, $i=0..n$, mốc phân biệt:
\[ P_n(x)=\sum_{i=0}^{n}y_iL_i(x),\quad L_i(x)=\prod_{j\ne i}\frac{x-x_j}{x_i-x_j}=\frac{1}{D_i}\cdot\frac{w_{n+1}(x)}{x-x_i},\quad D_i=\prod_{j\ne i}(x_i-x_j),\quad w_{n+1}(x)=\prod_{j=0}^{n}(x-x_j). \]

\begin{thuattoan}{Đa thức nội suy Lagrange dạng chính tắc}
\item[Input:] $x=(x_0,\dots,x_n)$, $y=(y_0,\dots,y_n)$.
\item[Output:] $a=(a_n,\dots,a_0)$ với $P_n(x)=\sum_l a_lx^l$.
\item[Bước 0:] Sắp các mốc tăng theo $x$; nếu hai mốc trùng nhau thì: cùng $y$ giữ một mốc, khác $y$ báo lỗi, dừng thuật toán.
\item[Bước 1:] Tính hệ số của $w_{n+1}(x)$ theo thuật toán Câu 5b.
\item[Bước 2:] Đặt $a_l:=0$, $l=0,\dots,n$.
\item[Bước 3:] Lặp $i=0,\dots,n$:
  \begin{con}
  \item[3.1:] Chia $w_{n+1}(x)$ cho $(x-x_i)$ theo thuật toán Câu 4a, được thương $q^{(i)}=(q^{(i)}_n,\dots,q^{(i)}_0)$.
  \item[3.2:] Tính $D_i:=\prod_{j\ne i}(x_i-x_j)$.
  \item[3.3:] Lặp $l=0,\dots,n$: tính $a_l:=a_l+\dfrac{y_i}{D_i}\,q^{(i)}_l$.
  \end{con}
\item[Bước 4:] Dừng thuật toán, kết quả là $a$.
\end{thuattoan}

\section{Câu 10. Thuật toán đa thức nội suy Newton tiến / lùi}

\pp{Công thức.} Mốc bất kỳ dùng tỷ sai phân; mốc cách đều $x_i=x_0+ih$ dùng sai phân $\Delta^ky_i$ và biến $t$:
\begin{center}\begin{adjustbox}{max width=\linewidth}\begin{tabular}{lll}\toprule
& Newton tiến & Newton lùi\\\midrule
Mốc bất kỳ & $\displaystyle\sum_{k=0}^nf[x_0..x_k]\prod_{i<k}(x-x_i)$ & $\displaystyle\sum_{k=0}^nf[x_{n-k}..x_n]\prod_{i<k}(x-x_{n-i})$\\
Mốc cách đều & $\displaystyle\sum_{k=0}^n\frac{\Delta^ky_0}{k!}t(t-1)\cdots(t-k+1),\ t=\frac{x-x_0}{h}$ & $\displaystyle\sum_{k=0}^n\frac{\nabla^ky_n}{k!}t(t+1)\cdots(t+k-1),\ t=\frac{x-x_n}{h}$\\\bottomrule
\end{tabular}\end{adjustbox}\end{center}
Trong bảng tam giác dưới $T_{i,k}$: Newton tiến lấy đường chéo $T_{k,k}$, Newton lùi lấy hàng cuối $T_{n,k}$ (vì $\nabla^ky_n=\Delta^ky_{n-k}$). Mốc cách đều: $f^{(j)}(x)\approx P^{(j)}(t)/h^j$.

\begin{thuattoan}{Newton tiến / lùi, mốc bất kỳ}
\item[Input:] $x=(x_0,\dots,x_n)$, $y=(y_0,\dots,y_n)$; chiều (tiến hoặc lùi); điểm $\bar x$; cấp đạo hàm $r$.
\item[Output:] vector hệ số $a$ của $P_n(x)$; $f_j\approx f^{(j)}(\bar x)$, $j=0..r$.

\section{Câu 15. Thuật toán Gauss I, Gauss II, Stirling, Bessel}

\section{Câu 18. Thuật toán cho bài toán nội suy ngược}

\subsection{a) Tất cả các khoảng cách ly của $f(x)=\bar y$}
\begin{thuattoan}{Khoảng cách ly}
\item[Input:] bảng $(x_i,y_i)$, $i=0..n$; giá trị $\bar y$.
\item[Output:] danh sách nghiệm tại mốc; danh sách khoảng cách ly $(x_i,x_{i+1})$.
\item[Bước 0:] Như Câu 6; nếu $\bar y\notin[\min y_i,\max y_i]$ thì báo ``không có nghiệm trong bảng'', dừng thuật toán.
\item[Bước 1:] Lặp $i=0,\dots,n$: tính $g_i:=y_i-\bar y$.
\item[Bước 2:] Lặp $i=0,\dots,n$:
  \begin{con}
  \item[2.1:] Nếu $g_i=0$ thì ghi $x_i$ vào danh sách nghiệm.
  \item[2.2:] Nếu $i<n$ và $g_i\,g_{i+1}<0$ thì ghi $(x_i,x_{i+1})$ vào danh sách khoảng cách ly.
  \end{con}
\item[Bước 3:] Dừng thuật toán, kết quả là hai danh sách trên.
\end{thuattoan}

\subsection{b) Các khoảng đơn điệu}
\begin{thuattoan}{Khoảng đơn điệu}
\item[Input:] bảng $(x_i,y_i)$, $i=0..n$, $n\ge1$.
\item[Output:] các đoạn $[x_a,x_b]$ kèm chiều tăng / giảm / hằng.
\item[Bước 0:] Như Câu 6.
\item[Bước 1:] Lặp $i=0,\dots,n-1$: tính $s_i:=\operatorname{sign}(y_{i+1}-y_i)$. Đặt $a:=0$.
\item[Bước 2:] Lặp $i=1,\dots,n-1$: nếu $s_i\ne s_{i-1}$ thì ghi đoạn $[x_a,x_i]$ với chiều $s_{i-1}$ và đặt $a:=i$.
\item[Bước 3:] Ghi đoạn $[x_a,x_n]$ với chiều $s_{n-1}$.\newline Dừng thuật toán, kết quả là danh sách các đoạn.
\end{thuattoan}

\subsection{c) Phương pháp hàm ngược}
\begin{thuattoan}{Hàm ngược}
\item[Input:] bảng $(x_i,y_i)$; $\bar y$; khoảng cách ly $(x_k,x_{k+1})$ nằm trong đoạn đơn điệu $[x_a,x_b]$ (câu a, b); số mốc $m\ge2$.
\item[Output:] nghiệm gần đúng $\bar x$.
\item[Bước 0:] Như câu a.
\item[Bước 1:] Đặt $L:=k$, $R:=k+1$.
\item[Bước 2:] Lặp $s=3,\dots,m$: kết nạp thêm một mốc:
  \begin{con}
  \item[TH1:] Nếu $L=a$ và $R=b$ thì dừng kết nạp (đoạn đơn điệu hết mốc).
  \item[TH2:] Nếu $L=a$ thì $R:=R+1$.
  \item[TH3:] Nếu $R=b$ thì $L:=L-1$.
  \item[TH4:] Còn lại, nếu $|y_{L-1}-\bar y|\le|y_{R+1}-\bar y|$ thì $L:=L-1$, trái lại $R:=R+1$.
  \end{con}
\item[Bước 3:] Đặt $u_s:=y_{L+s}$, $v_s:=x_{L+s}$, $s=0..R-L$; nếu đoạn giảm thì đảo thứ tự để $u_0<u_1<\cdots$.
\item[Bước 4:] Lập bảng tỷ sai phân của $v$ theo $u$ (bước 1, Newton mốc bất kỳ — Câu 10); đặt $c_j:=T_{j,j}$, $j=0..R-L$.
\item[Bước 5:] Đặt $\omega:=1$, $S:=c_0$. Lặp $j=1,\dots,R-L$: tính $\omega:=\omega\,(\bar y-u_{j-1})$ và $S:=S+c_j\,\omega$.
\item[Bước 6:] Đặt $\bar x:=S$.\newline Dừng thuật toán, kết quả là $\bar x$ (số hạng cuối $|c_{R-L}\,\omega|$ là ước lượng sai số).
\end{thuattoan}

\subsection{d) Phương pháp lặp}
\begin{thuattoan}{Lặp Newton tiến / lùi}
\item[Input:] bảng $(x_i,y_i)$, $i=0..n$, cách đều bước $h$; $\bar y$; khoảng cách ly $(x_k,x_{k+1})$; số mốc $m$; sai số $\varepsilon$; số lần lặp tối đa $N$.
\item[Output:] nghiệm gần đúng $\bar x$, hoặc thông báo ``Không hội tụ''.
\item[Bước 0:] Như câu a; nếu mốc không cách đều thì dùng phương pháp hàm ngược. Nếu $k+m-1>n$ và $k+2-m<0$ thì đặt $m:=\max(n-k+1,\,k+2)$.
\item[Bước 1:] Chọn chiều: nếu $k+m-1\le n$ thì lặp tiến, đặt $s:=k$; trái lại lặp lùi, đặt $s:=k+1$.
\item[Bước 2:] Lập bảng sai phân $T$ (bước 1, Newton mốc cách đều — Câu 10, đánh số lại $0..m-1$): nếu tiến thì của các mốc $x_k,\dots,x_{k+m-1}$ và đặt $d_j:=T_{j,j}=\Delta^jy_k$; nếu lùi thì của các mốc $x_{k+2-m},\dots,x_{k+1}$ và đặt $d_j:=T_{m-1,j}=\nabla^jy_{k+1}$, $j=0..m-1$.
\item[Bước 3:] Đặt $A:=\dfrac{\bar y-d_0}{d_1}$, $B_j:=\dfrac{d_j}{j!\,d_1}$; nếu tiến thì $\varphi(t):=A-\sum_{j=2}^{m-1}B_j\,t(t-1)\cdots(t-j+1)$; nếu lùi thì $\varphi(t):=A-\sum_{j=2}^{m-1}B_j\,t(t+1)\cdots(t+j-1)$.
\item[Bước 4:] Đặt $t_0:=A$, $l:=0$.
\item[Bước 5:] Tính $t_{l+1}:=\varphi(t_l)$.
\item[Bước 6:] Nếu $h\,|t_{l+1}-t_l|<\varepsilon$ thì đặt $\bar x:=x_s+t_{l+1}h$, dừng thuật toán.
\item[Bước 7:] Nếu $l+1=N$ thì báo ``Không hội tụ'', dừng thuật toán; trái lại đặt $l:=l+1$, quay lại bước 5.
\end{thuattoan}

\section{Câu 20. Hàm ghép trơn (spline)}

\subsection{a) Xây dựng công thức}
Spline cấp $p$: $S(x)=S_k(x)$ trên $[x_k,x_{k+1}]$, $S_k$ đa thức bậc $p$, $S\in C^{p-1}[x_0,x_n]$, $S(x_i)=y_i$; $h_k=x_{k+1}-x_k$; cần thêm $p-1$ điều kiện biên.

\pp{Spline cấp 3.} Đặt $\sigma_k=S''(x_k)$. $S_k''$ bậc nhất nên $S_k''(x)=\dfrac{\sigma_k(x_{k+1}-x)+\sigma_{k+1}(x-x_k)}{h_k}$; tích phân hai lần:
\[ S_k(x)=\frac{\sigma_k(x_{k+1}-x)^3+\sigma_{k+1}(x-x_k)^3}{6h_k}+\alpha_k(x-x_k)+\beta_k(x_{k+1}-x). \]
$S_k(x_k)=y_k\Rightarrow\beta_k=\dfrac{y_k}{h_k}-\dfrac{\sigma_kh_k}{6}$; \ $S_k(x_{k+1})=y_{k+1}\Rightarrow\alpha_k=\dfrac{y_{k+1}}{h_k}-\dfrac{\sigma_{k+1}h_k}{6}$. Đặt $s_k=\dfrac{y_{k+1}-y_k}{h_k}$:
\[ S_k'(x)=\frac{-\sigma_k(x_{k+1}-x)^2+\sigma_{k+1}(x-x_k)^2}{2h_k}+s_k-\frac{(\sigma_{k+1}-\sigma_k)h_k}{6}, \]
\[ S_k'(x_{k+1})=s_k+\frac{h_k}{6}\sigma_k+\frac{h_k}{3}\sigma_{k+1},\qquad S_{k+1}'(x_{k+1})=s_{k+1}-\frac{h_{k+1}}{3}\sigma_{k+1}-\frac{h_{k+1}}{6}\sigma_{k+2}. \]
Trơn cấp 1 tại $x_{k+1}$ cho $n-1$ phương trình ($k=0..n-2$):
\[ \boxed{\frac{h_k}{6}\sigma_k+\frac{h_k+h_{k+1}}{3}\sigma_{k+1}+\frac{h_{k+1}}{6}\sigma_{k+2}=s_{k+1}-s_k} \]
cộng 2 điều kiện biên: $S''(x_0)=A\Rightarrow\sigma_0=A$; $S'(x_0)=A\Rightarrow\frac{h_0}{3}\sigma_0+\frac{h_0}{6}\sigma_1=s_0-A$; $S''(x_n)=B\Rightarrow\sigma_n=B$; $S'(x_n)=B\Rightarrow\frac{h_{n-1}}{6}\sigma_{n-1}+\frac{h_{n-1}}{3}\sigma_n=B-s_{n-1}$. Hệ ba đường chéo, chéo trội. Khai triển được dạng chính tắc $S_k=a_kx^3+b_kx^2+c_kx+d_k$:
\[ a_k=\frac{\sigma_{k+1}-\sigma_k}{6h_k},\ b_k=\frac{\sigma_kx_{k+1}-\sigma_{k+1}x_k}{2h_k},\ c_k=\frac{-\sigma_kx_{k+1}^2+\sigma_{k+1}x_k^2}{2h_k}+s_k-\frac{(\sigma_{k+1}-\sigma_k)h_k}{6}, \]
\[ d_k=\frac{\sigma_kx_{k+1}^3-\sigma_{k+1}x_k^3}{6h_k}+\frac{y_kx_{k+1}-y_{k+1}x_k}{h_k}+\frac{(\sigma_{k+1}x_k-\sigma_kx_{k+1})h_k}{6}. \]

\pp{Spline cấp 4.} Đặt $m_k=S'(x_k)$, $\sigma_k=S''(x_k)$, $M_k=S'''(x_k)$ ($S\in C^3$ nên các giá trị này chung cho hai mảnh kề). $S_k'''$ bậc nhất: $S_k'''(x)=M_k+\frac{M_{k+1}-M_k}{h_k}u$, $u=x-x_k$. Tích phân ba lần với giá trị ban đầu tại $x_k$ (khai triển Taylor):
\[ S_k(x)=y_k+m_ku+\frac{\sigma_k}{2}u^2+\frac{M_k}{6}u^3+\frac{M_{k+1}-M_k}{24h_k}u^4,\qquad u=x-x_k. \]
Liên tục của $S,S',S''$ tại $x_{k+1}$ ($S'''$ liên tục sẵn), $k=0..n-1$:
\[ \boxed{\begin{aligned}
&y_k+m_kh_k+\frac{\sigma_k}{2}h_k^2+\frac{3M_k+M_{k+1}}{24}h_k^3=y_{k+1},\\
&m_k+\sigma_kh_k+\frac{2M_k+M_{k+1}}{6}h_k^2=m_{k+1},\\
&\sigma_k+\frac{M_k+M_{k+1}}{2}h_k=\sigma_{k+1}.
\end{aligned}} \]
$3n$ phương trình, $3(n+1)$ ẩn $(m_k,\sigma_k,M_k)$ $\Rightarrow$ cần 3 điều kiện biên (vd $S'(x_0),S''(x_0),S'(x_n)$). Dạng chính tắc: khai triển $u=x-x_k$ bằng Horner nhân.

\subsection{b) Thuật toán spline cấp $p\le4$}
Bước 0 dùng chung: sắp các mốc tăng theo $x$; nếu hai mốc trùng nhau thì báo lỗi, dừng thuật toán; nếu số điều kiện biên khác $p-1$ hoặc có điều kiện không đặt tại $x_0$, $x_n$ thì báo lỗi, dừng thuật toán.

\begin{thuattoan}{Spline cấp 1}
\item[Input:] $(x_i,y_i)$, $i=0..n$.
\item[Output:] $S_k(x)=a_kx+b_k$, $k=0..n-1$.
\item[Bước 0:] Như trên.
\item[Bước 1:] Lặp $k=0,\dots,n-1$: tính $h_k:=x_{k+1}-x_k$, $a_k:=\dfrac{y_{k+1}-y_k}{h_k}$, $b_k:=\dfrac{y_kx_{k+1}-y_{k+1}x_k}{h_k}$.\newline Dừng thuật toán, kết quả là các $S_k$.
\end{thuattoan}

\begin{thuattoan}{Spline cấp 2}
\item[Input:] $(x_i,y_i)$, $i=0..n$; một điều kiện biên $S'(x_0)=A$ hoặc $S'(x_n)=B$.
\item[Output:] $S_k(x)=a_kx^2+b_kx+c_k$, $k=0..n-1$.
\item[Bước 0:] Như trên.
\item[Bước 1:] Lặp $k=0,\dots,n-1$: tính $h_k:=x_{k+1}-x_k$ và $\gamma_k:=\dfrac{2(y_{k+1}-y_k)}{h_k}$.
\item[Bước 2:] Tính $m_k=S'(x_k)$:
  \begin{con}
  \item[TH1:] Nếu cho $S'(x_0)=A$ thì đặt $m_0:=A$; lặp $k=0,\dots,n-1$: tính $m_{k+1}:=\gamma_k-m_k$.
  \item[TH2:] Nếu cho $S'(x_n)=B$ thì đặt $m_n:=B$; lặp $k=n-1,\dots,0$: tính $m_k:=\gamma_k-m_{k+1}$.
  \end{con}
\item[Bước 3:] Lặp $k=0,\dots,n-1$: tính
  \[ a_k:=\frac{m_{k+1}-m_k}{2h_k},\ \ b_k:=\frac{m_kx_{k+1}-m_{k+1}x_k}{h_k},\ \ c_k:=\frac{-m_kx_{k+1}^2+m_{k+1}x_k^2}{2h_k}+y_k+\frac{m_kh_k}{2}. \]
  Dừng thuật toán, kết quả là các $S_k$.
\end{thuattoan}

\begin{thuattoan}{Spline cấp 3}
\item[Input:] $(x_i,y_i)$, $i=0..n$; hai điều kiện biên dạng $S'(\xi)=v$ hoặc $S''(\xi)=v$, $\xi\in\{x_0,x_n\}$.
\item[Output:] $S_k(x)=a_kx^3+b_kx^2+c_kx+d_k$, $k=0..n-1$.
\item[Bước 0:] Như trên.
\item[Bước 1:] Lặp $k=0,\dots,n-1$: tính $h_k:=x_{k+1}-x_k$ và $s_k:=\dfrac{y_{k+1}-y_k}{h_k}$.
\item[Bước 2:] Lập hệ $n+1$ phương trình ẩn $\sigma_0,\dots,\sigma_n$: $n-1$ phương trình trơn (khung, câu a) và 2 phương trình biên (câu a).
\item[Bước 3:] Nếu mỗi đầu có một điều kiện thì hệ có dạng ba đường chéo $\ell_i\sigma_{i-1}+d_i\sigma_i+u_i\sigma_{i+1}=r_i$; giải bằng truy đuổi:
  \begin{con}
  \item[3.1:] Đặt $P_0:=-u_0/d_0$, $Q_0:=r_0/d_0$. Lặp $i=1,\dots,n$: tính $\mu:=d_i+\ell_iP_{i-1}$ (nếu $\mu=0$ thì báo hệ suy biến, dừng thuật toán), $P_i:=-u_i/\mu$, $Q_i:=(r_i-\ell_iQ_{i-1})/\mu$.
  \item[3.2:] Đặt $\sigma_n:=Q_n$. Lặp $i=n-1,\dots,0$: tính $\sigma_i:=P_i\sigma_{i+1}+Q_i$. Sang bước 5.
  \end{con}
\item[Bước 4:] Nếu hai điều kiện cùng một đầu thì giải hệ bằng khử Gauss chọn trụ (bước 3, Câu 22a).
\item[Bước 5:] Lặp $k=0,\dots,n-1$: tính $a_k,b_k,c_k,d_k$ theo câu a.\newline Dừng thuật toán, kết quả là các $S_k$.
\end{thuattoan}

\begin{thuattoan}{Spline cấp 4}

\section{Câu 22. Thuật toán phương pháp bình phương tối thiểu}

\subsection{a) Hàm thực nghiệm với tập cơ sở $\{\varphi_j\}_{j=1}^m$ và sai số trung bình phương}
\begin{thuattoan}{Bình phương tối thiểu tuyến tính theo tham số}
\item[Input:] $(x_i,y_i)$, $i=1..n$; cơ sở $\varphi_1,\dots,\varphi_m$.
\item[Output:] hệ số $a=(a_1,\dots,a_m)$ của $g=\sum_ja_j\varphi_j$; sai số trung bình phương $\sigma$.
\item[Bước 0:] Loại các điểm làm $\varphi_j(x_i)$ không xác định; nếu còn $n<m$ điểm thì báo lỗi, dừng thuật toán.
\item[Bước 1:] Lặp $i=1,\dots,n$; $j=1,\dots,m$: tính $\Phi_{ij}:=\varphi_j(x_i)$.
\item[Bước 2:] Lặp $j,k=1,\dots,m$: tính $M_{jk}:=\sum_{i=1}^n\Phi_{ij}\Phi_{ik}$ và $b_j:=\sum_{i=1}^n\Phi_{ij}y_i$.
\item[Bước 3:] Giải hệ chuẩn $Ma=b$ bằng khử Gauss chọn trụ (hàng $L_j$ của ma trận mở rộng $[M\,|\,b]$):
  \begin{con}
  \item[3.1:] Lặp $c=1,\dots,m$: chọn $p\ge c$ có $|M_{pc}|$ lớn nhất (nếu $M_{pc}=0$ thì báo ``ma trận suy biến'', dừng thuật toán); đổi chỗ $L_p$, $L_c$; lặp $l=c+1,\dots,m$: biến đổi $L_l:=L_l-\dfrac{M_{lc}}{M_{cc}}L_c$.
  \item[3.2:] Lặp $c=m,\dots,1$: tính $a_c:=\Big(b_c-\sum_{l>c}M_{cl}a_l\Big)/M_{cc}$.
  \end{con}
\item[Bước 4:] Tính $\sigma:=\sqrt{\frac1n\sum_{i=1}^n\big(\sum_ja_j\Phi_{ij}-y_i\big)^2}$.\newline Dừng thuật toán, kết quả là $a$ và $\sigma$.
\end{thuattoan}

\subsection{b) Dạng $y=a\,e^{b_1\varphi_1(x)+b_2\varphi_2(x)}$ và $y=a\,x^b$}
\begin{thuattoan}{Dạng mũ $y=a\,e^{b_1\varphi_1(x)+b_2\varphi_2(x)}$}
\item[Input:] $(x_i,y_i)$, $i=1..n$; cơ sở $\varphi_1,\varphi_2$ (không phải hàm hằng).
\item[Output:] $a$, $b_1$, $b_2$ và $\sigma$.
\item[Bước 0:] Như câu a; loại các điểm có $y_i=0$ hoặc trái dấu với đa số.
\item[Bước 1:] Đặt $s:=\operatorname{sign}y_i$. Lặp $i=1,\dots,n$: tính $z_i:=\ln(s\,y_i)$.
\item[Bước 2:] Tìm $(c_0,c_1,c_2)$ theo câu a với dữ liệu $(x_i,z_i)$ và cơ sở $(1,\varphi_1,\varphi_2)$.
\item[Bước 3:] Đặt $a:=s\,e^{c_0}$, $b_1:=c_1$, $b_2:=c_2$; tính $\sigma:=\sqrt{\frac1n\sum_i\big(a\,e^{b_1\varphi_1(x_i)+b_2\varphi_2(x_i)}-y_i\big)^2}$.\newline Dừng thuật toán, kết quả là $a$, $b_1$, $b_2$, $\sigma$.
\end{thuattoan}

\begin{thuattoan}{Dạng luỹ thừa $y=a\,x^b$}
\item[Input:] $(x_i,y_i)$, $i=1..n$.
\item[Output:] $a$, $b$ và $\sigma$.
\item[Bước 0:] Như câu a; loại các điểm có $y_i=0$, trái dấu với đa số, hoặc $x_i\le0$.
\item[Bước 1:] Đặt $s:=\operatorname{sign}y_i$. Lặp $i=1,\dots,n$: tính $X_i:=\ln x_i$ và $z_i:=\ln(s\,y_i)$.
\item[Bước 2:] Tìm $(c_0,c_1)$ theo câu a với dữ liệu $(X_i,z_i)$ và cơ sở $(1,X)$.
\item[Bước 3:] Đặt $a:=s\,e^{c_0}$, $b:=c_1$; tính $\sigma:=\sqrt{\frac1n\sum_i(a\,x_i^b-y_i)^2}$.\newline Dừng thuật toán, kết quả là $a$, $b$, $\sigma$.
\end{thuattoan}

\begin{thuattoan}{Hiệu chỉnh Gauss--Newton (cho hai dạng trên)}
\item[Input:] $(x_i,y_i)$; dấu $s$; ma trận cơ sở $\Phi$ của $(1,\varphi_1,\varphi_2)$ (dạng luỹ thừa: $(1,\ln x)$); sai số $\varepsilon$; số lần lặp tối đa $N$.
\item[Output:] $c$ cực tiểu $\sum_i\big(s\,e^{(\Phi c)_i}-y_i\big)^2$; $a$, $b_j$, $\sigma$; hoặc thông báo ``Không hội tụ''.
\item[Bước 1:] Tìm $c$ theo câu a có trọng số $w_i:=y_i^2$ với dữ liệu $z_i:=\ln(s\,y_i)$ (tức $M_{jk}:=\sum_iw_i\Phi_{ij}\Phi_{ik}$, $b_j:=\sum_iw_i\Phi_{ij}z_i$). Đặt $l:=0$.
\item[Bước 2:] Lặp $i=1,\dots,n$: tính $g_i:=s\,e^{(\Phi c)_i}$, $\tilde z_i:=\ln(s\,g_i)+\dfrac{y_i-g_i}{g_i}$ và $w_i:=g_i^2$.
\item[Bước 3:] Tìm $c^{\text{mới}}$ như bước 1 với trọng số $w_i$ và dữ liệu $\tilde z_i$.
\item[Bước 4:] Nếu $\max|c^{\text{mới}}-c|<\varepsilon\,(1+\max|c|)$ thì đặt $c:=c^{\text{mới}}$, sang bước 6.
\item[Bước 5:] Nếu $l+1=N$ thì báo ``Không hội tụ'', dừng thuật toán; trái lại đặt $c:=c^{\text{mới}}$, $l:=l+1$, quay lại bước 2.
\item[Bước 6:] Đặt $a:=s\,e^{c_0}$, $b_j:=c_j$; tính $\sigma:=\sqrt{\frac1n\sum_i\big(s\,e^{(\Phi c)_i}-y_i\big)^2}$.\newline Dừng thuật toán, kết quả là $a$, $b_j$, $\sigma$.
\end{thuattoan}
