\section{Câu 3. Mốc nội suy tối ưu}

\subsection{a) Xây dựng công thức $n$ mốc nội suy tối ưu trên $[-1,1]$ và trên $[a,b]$}

\pp{Đặt bài toán.} Nội suy $f$ bởi đa thức $P$ bậc $\le n-1$ qua $n$ mốc $x_0,\dots,x_{n-1}\in[a,b]$ có sai số
\[ f(x)-P(x)=\frac{f^{(n)}(\xi)}{n!}\,w_n(x),\qquad w_n(x)=\prod_{i=0}^{n-1}(x-x_i). \]
Phần phụ thuộc vào cách chọn mốc là $w_n$ — một đa thức bậc $n$ có hệ số cao nhất bằng 1 (đa thức monic). Mốc \emph{tối ưu} là mốc làm $\max_{x\in[a,b]}|w_n(x)|$ nhỏ nhất.

\pp{Đa thức Chebyshev.} $T_n(t)=\cos(n\arccos t)$, $t\in[-1,1]$. Từ $\cos((n+1)\theta)+\cos((n-1)\theta)=2\cos\theta\cos n\theta$ có $T_{n+1}(t)=2tT_n(t)-T_{n-1}(t)$, $T_0=1$, $T_1=t$, nên quy nạp được: $T_n$ là đa thức bậc $n$ với hệ số cao nhất $2^{n-1}$ ($n\ge1$). Ngoài ra
\begin{itemize}
\item $|T_n(t)|\le1$ trên $[-1,1]$ và $T_n(\tau_k)=(-1)^k$ tại $n+1$ điểm $\tau_k=\cos\frac{k\pi}{n}$, $k=0..n$ (luân phiên dấu);
\item $T_n$ có $n$ nghiệm phân biệt $t_i=\cos\dfrac{(2i+1)\pi}{2n}$, $i=0..n-1$, đều thuộc $(-1,1)$.
\end{itemize}

\pp{Định lý.} Mọi đa thức monic $p$ bậc $n$ thoả $\max_{[-1,1]}|p|\ge2^{1-n}$; dấu bằng đạt tại $\tilde T_n=2^{1-n}T_n$.

\emph{Chứng minh.} $\tilde T_n$ monic và $\max|\tilde T_n|=2^{1-n}$. Giả sử có $p$ monic bậc $n$ với $\max|p|<2^{1-n}$. Đặt $q=\tilde T_n-p$: hai số hạng $t^n$ triệt tiêu nên $\deg q\le n-1$. Tại $\tau_k$: $\tilde T_n(\tau_k)=(-1)^k2^{1-n}$ còn $|p(\tau_k)|<2^{1-n}$, nên $q(\tau_k)$ cùng dấu $(-1)^k$, $k=0..n$. Vậy $q$ đổi dấu $n$ lần, có ít nhất $n$ nghiệm trong $(-1,1)$, mà $\deg q\le n-1$ $\Rightarrow q\equiv0\Rightarrow p=\tilde T_n$, mâu thuẫn với $\max|p|<2^{1-n}$. $\square$

\pp{Trên $[-1,1]$.} $w_n$ monic nên $\max|w_n|$ nhỏ nhất khi $w_n=\tilde T_n$, tức các mốc là nghiệm của $T_n$:
\[ \boxed{t_i=\cos\frac{(2i+1)\pi}{2n},\ i=0,\dots,n-1,\qquad \min\max_{[-1,1]}|w_n|=\frac{1}{2^{n-1}}.} \]

\pp{Trên $[a,b]$.} Đổi biến $x=\dfrac{a+b}{2}+\dfrac{b-a}{2}t$ ánh xạ $[-1,1]$ lên $[a,b]$. Khi đó $x-x_i=\frac{b-a}{2}(t-t_i)$ nên $w_n(x)=\left(\frac{b-a}{2}\right)^n\prod(t-t_i)$; bài toán trên $[a,b]$ tương đương bài toán trên $[-1,1]$:
\[ \boxed{x_i=\frac{a+b}{2}+\frac{b-a}{2}\cos\frac{(2i+1)\pi}{2n},\qquad \min\max_{[a,b]}|w_n|=\left(\frac{b-a}{2}\right)^n\frac{1}{2^{n-1}}=\frac{(b-a)^n}{2^{2n-1}},} \]
và khi đó $|f(x)-P(x)|\le\dfrac{M_n}{n!}\cdot\dfrac{(b-a)^n}{2^{2n-1}}$, $M_n=\max_{[a,b]}|f^{(n)}|$.

\pp{Sắp tăng dần.} $\cos$ giảm trên $[0,\pi]$ nên $t_i$ ở trên giảm theo $i$. Dùng $t_i=-\cos\frac{(2i+1)\pi}{2n}=\cos\frac{(2(n-1-i)+1)\pi}{2n}$ (cùng tập nghiệm, đánh số ngược) thì $t_0<t_1<\dots<t_{n-1}$.

\subsection{b) Thuật toán xác định $n$ mốc tối ưu sắp tăng dần trên $[a,b]$}
\begin{thuattoan}{Mốc nội suy tối ưu}
\item[Input:] số mốc $n\in\mathbb N$, $n\ge1$; đoạn $[a,b]$.
\item[Output:] vector $x=(x_0,\dots,x_{n-1})$, $x_0<\dots<x_{n-1}$; $W=\max_{[a,b]}|w_n|$.
\item[Bước 0 — Xử lý dữ liệu:] chuẩn hoá số (dấu trừ Unicode, dấu phẩy thập phân); $n$ phải nguyên $\ge1$; $a=b$ $\Rightarrow$ dừng, báo lỗi; $a>b$ $\Rightarrow$ đổi chỗ $a\leftrightarrow b$ và báo.
\item[Bước 1 — Đổi biến:] $p:=\frac{a+b}{2}$, $q:=\frac{b-a}{2}$.
\item[Bước 2 — Với $i=0,1,\dots,n-1$:] 2.1: $t_i:=-\cos\dfrac{(2i+1)\pi}{2n}$; \quad 2.2: $x_i:=p+q\,t_i$.
\item[Bước 3 — Độ lệch nhỏ nhất:] $W:=\dfrac{(b-a)^n}{2^{2n-1}}$ (tính qua $\log$ khi $n$ lớn để không tràn số).
\item[Bước 4 — Kết thúc:] trả về $x$, $W$.
\end{thuattoan}

\section{Câu 4. Sơ đồ Horner: giá trị, phép chia, đạo hàm}

\subsection{a) Giá trị $P_n(c)$ \quad b) Chia $P_n(x)$ cho $(x-c)$}
\pp{Cơ sở.} Viết $P_n(x)=(x-c)Q_{n-1}(x)+r$, $Q_{n-1}=b_{n-1}x^{n-1}+\dots+b_0$. Đồng nhất hệ số: $a_n=b_{n-1}$, $a_i=b_{i-1}-c\,b_i$ ($1\le i\le n-1$), $a_0=r-c\,b_0$; và $r=P_n(c)$. Do đó tính giá trị chính là lấy số dư của phép chia (một sơ đồ cho cả a và b), chỉ tốn $n$ phép nhân, $n$ phép cộng.
\begin{thuattoan}{Horner chia (giá trị, thương, số dư)}
\item[Input:] $a=(a_n,\dots,a_0)$, $a_n\neq0$; số $c$.
\item[Output:] thương $b=(b_{n-1},\dots,b_0)$ và số dư $r=P_n(c)$.
\item[Bước 0 — Xử lý dữ liệu:] đọc dãy số (khoảng trắng/\texttt{;}, dấu trừ Unicode, dấu phẩy thập phân, bỏ ô chữ và báo); NaN/$\pm\infty$ $\Rightarrow$ dừng; bỏ các hệ số 0 ở đầu tới khi $a_n\ne0$ (mọi hệ số 0 $\Rightarrow$ $P\equiv0$, dừng).
\item[Bước 1 — Bậc 0:] nếu $n=0$: $Q\equiv0$, $r:=a_0$, dừng.
\item[Bước 2 — Khởi tạo:] $b_{n-1}:=a_n$.
\item[Bước 3 — Với $i=n-1,\dots,1$:] $b_{i-1}:=a_i+c\,b_i$.
\item[Bước 4 — Số dư:] $r:=a_0+c\,b_0$; trả về $b$, $r$. \quad (Chỉ cần giá trị: chỉ giữ $r$.)
\end{thuattoan}

\subsection{c) Đạo hàm cấp $k$ của $P_n$ tại $c$}
\pp{Cơ sở.} Chia liên tiếp cho $(x-c)$: $P_n=(x-c)Q^{(1)}+r_0$, $Q^{(1)}=(x-c)Q^{(2)}+r_1$, … nên
$P_n(x)=r_0+r_1(x-c)+r_2(x-c)^2+\dots+r_n(x-c)^n$. Đây là khai triển Taylor tại $c$, do đó
\[ r_j=\frac{P_n^{(j)}(c)}{j!}\quad\Longrightarrow\quad P_n^{(j)}(c)=j!\,r_j. \]
\begin{thuattoan}{Đạo hàm cấp $0..k$ tại $c$ (Horner chia lặp)}
\item[Input:] $a=(a_n,\dots,a_0)$; $c$; cấp $k\in\mathbb N$.
\item[Output:] $r=(r_0,\dots,r_K)$, $K=\min(k,n)$; $v=(v_0,\dots,v_k)$, $v_j=P_n^{(j)}(c)$.
\item[Bước 0 — Xử lý dữ liệu:] như câu a; $k$ phải nguyên $\ge0$.
\item[Bước 1 — Khởi tạo:] $K:=\min(k,n)$; vector làm việc $b_i:=a_i$, $i=0..n$.
\item[Bước 2 — Với $j=0,\dots,K$:] 2.1 (chia cho $x-c$): với $i=n-1,\dots,j$: $b_i:=b_i+c\,b_{i+1}$; \quad 2.2: $r_j:=b_j$, $v_j:=j!\,r_j$.
\item[Bước 3 — Cấp lớn hơn bậc:] $v_j:=0$ với $j=K+1..k$; trả về $r$, $v$.
\end{thuattoan}

\subsection{d) Áp dụng: $P(x)=3x^3-2x-1$, $a=(3;\,0;\,-2;\,-1)$}
\pp{Phương pháp.} Dùng sơ đồ Horner chia lặp (câu c): lần chia thứ nhất cho ngay $P(c)$ (số dư) và thương (câu a, b); các lần chia sau cho $P^{(j)}(c)=j!\,r_j$, $j=1,2,3$ ($P$ bậc 3 nên $P^{(j)}\equiv0$ với $j\ge4$). \section{Câu 5. Phép nhân Horner, đa thức $w_{n+1}(x)$}

\subsection{a) Nhân $P_n(x)$ với $(x-c)$}
\pp{Cơ sở.} $(x-c)\sum_{i=0}^na_ix^i=\sum_{i=0}^{n}a_ix^{i+1}-\sum_{i=0}^nc\,a_ix^i$ nên hệ số $d_i$ của $x^i$ trong tích là $d_{n+1}=a_n$, $d_i=a_{i-1}-c\,a_i$ ($1\le i\le n$), $d_0=-c\,a_0$.
\begin{thuattoan}{Horner nhân}
\item[Input:] $a=(a_n,\dots,a_0)$; $c$. \quad\textbf{Output:} $d=(d_{n+1},\dots,d_0)$ — hệ số của $P_n(x)(x-c)$.
\item[Bước 0 — Xử lý dữ liệu:] như câu 4a.
\item[Bước 1:] $d_{n+1}:=a_n$. \quad\textbf{Bước 2 — Với $i=n,\dots,1$:} $d_i:=a_{i-1}-c\,a_i$. \quad\textbf{Bước 3:} $d_0:=-c\,a_0$; trả về $d$.
\end{thuattoan}

\subsection{b) Hệ số của $w_{n+1}(x)=\prod_{k=0}^{n}(x-x_k)$}
\begin{thuattoan}{Bảng tích (nhân Horner lặp)}
\item[Input:] mốc $x_0,\dots,x_n$. \quad\textbf{Output:} $p=(1,p_n,\dots,p_0)$, $w_{n+1}(x)=x^{n+1}+p_nx^n+\dots+p_0$.
\item[Bước 0 — Xử lý dữ liệu:] đọc dãy mốc như câu 4a; cần ít nhất 1 mốc.
\item[Bước 1 — Khởi tạo:] đa thức hằng $p_0:=1$.
\item[Bước 2 — Với $i=0,\dots,n$] (nhân thêm $(x-x_i)$ theo câu a, làm tại chỗ):
2.1: $p_{i+1}:=p_i$; \quad 2.2: với $l=i,\dots,1$: $p_l:=p_{l-1}-x_i\,p_l$; \quad 2.3: $p_0:=-x_i\,p_0$.
\item[Bước 3 — Kết thúc:] trả về $p$ (luôn có hệ số cao nhất bằng 1).
\end{thuattoan}

\section{Câu 6. Thuật toán đa thức nội suy Lagrange}

\pp{Công thức.} Với bộ điểm $(x_i,y_i)$, $i=0..n$, mốc phân biệt:
\[ P_n(x)=\sum_{i=0}^{n}y_iL_i(x),\quad L_i(x)=\prod_{j\ne i}\frac{x-x_j}{x_i-x_j}=\frac{1}{D_i}\cdot\frac{w_{n+1}(x)}{x-x_i},\quad D_i=\prod_{j\ne i}(x_i-x_j),\quad w_{n+1}(x)=\prod_{j=0}^{n}(x-x_j), \]
sai số $|f(x)-P_n(x)|\le\dfrac{M_{n+1}}{(n+1)!}|w_{n+1}(x)|$. Ý tưởng: lập $w_{n+1}$ một lần (bảng tích), mỗi $\frac{w_{n+1}(x)}{x-x_i}$ là \emph{thương} của phép chia Horner (dư bằng 0).

\begin{thuattoan}{Đa thức nội suy Lagrange dạng chính tắc}
\item[Input:] $x=(x_0,\dots,x_n)$, $y=(y_0,\dots,y_n)$.
\item[Output:] $a=(a_n,\dots,a_0)$ với $P_n(x)=\sum a_lx^l$; các $D_i$.
\item[Bước 0 — Xử lý dữ liệu:] đọc hàng $x$, hàng $y$ (bỏ nhãn, chuẩn hoá dấu trừ Unicode và dấu phẩy thập phân, bỏ ký tự lạ); hai hàng khác số ô $\Rightarrow$ dừng; ô rỗng/chữ/NaN/$\pm\infty$ $\Rightarrow$ loại cả cặp và báo; mốc trùng: cùng $y$ giữ 1, khác $y$ ít (chênh $\le10^{-3}\max(1,\max|y|)$) lấy trung bình, khác nhiều $\Rightarrow$ dừng (không có đa thức qua cả hai điểm); sắp tăng theo $x$.
\item[Bước 1 — Bảng tích:] hệ số $p$ của $w_{n+1}$ theo câu 5b.
\item[Bước 2 — Khởi tạo:] $a_l:=0$, $l=0..n$.
\item[Bước 3 — Với $i=0..n$:] 3.1 (bảng chia): chia $w_{n+1}$ cho $(x-x_i)$ bằng Horner (câu 4b) $\to$ thương $q^{(i)}=(q^{(i)}_n,\dots,q^{(i)}_0)$; \ 3.2: $D_i:=\prod_{j\ne i}(x_i-x_j)$; \ 3.3 (cộng dồn): $a_l:=a_l+\frac{y_i}{D_i}q^{(i)}_l$, $l=0..n$.
\item[Bước 4 — Kết thúc:] trả về $a$, $D$.
\end{thuattoan}
\begin{thuattoan}{Giá trị $P_n(\bar x)$ không lập đa thức}
\item[Bước 1:] nếu $\bar x=x_m$: trả về $y_m$. \quad\textbf{Bước 2:} $w:=\prod_i(\bar x-x_i)$. \quad\textbf{Bước 3:} $S:=\sum_i\dfrac{y_i}{D_i(\bar x-x_i)}$. \quad\textbf{Bước 4:} $P_n(\bar x):=w\,S$.
\end{thuattoan}

\section{Câu 10. Thuật toán đa thức nội suy Newton tiến / lùi}

\pp{Công thức.} Mốc bất kỳ (TH1) dùng tỷ sai phân; mốc cách đều $x_i=x_0+ih$ (TH2) dùng sai phân $\Delta^ky_i$ và biến $t$:
\begin{center}\begin{tabular}{lll}\toprule
& Newton tiến & Newton lùi\\\midrule
TH1 & $\displaystyle\sum_{k=0}^nf[x_0..x_k]\prod_{i<k}(x-x_i)$ & $\displaystyle\sum_{k=0}^nf[x_{n-k}..x_n]\prod_{i<k}(x-x_{n-i})$\\
TH2 & $\displaystyle\sum_{k=0}^n\frac{\Delta^ky_0}{k!}t(t-1)\cdots(t-k+1),\ t=\frac{x-x_0}{h}$ & $\displaystyle\sum_{k=0}^n\frac{\nabla^ky_n}{k!}t(t+1)\cdots(t+k-1),\ t=\frac{x-x_n}{h}$\\\bottomrule
\end{tabular}\end{center}
Hai bảng dùng chung một sơ đồ: bảng tam giác dưới $T_{i,k}$; Newton tiến lấy đường chéo $T_{k,k}$, Newton lùi lấy hàng cuối $T_{n,k}$ (vì $\nabla^ky_n=\Delta^ky_{n-k}$). Mốc cách đều: $f^{(j)}(x)=P^{(j)}(t)/h^j$.

\begin{thuattoan}{Đa thức nội suy Newton tiến / lùi}
\item[Input:] $x=(x_0..x_n)$, $y=(y_0..y_n)$; chiều (tiến/lùi); (tuỳ chọn) điểm $\bar x$, cấp $k$.
\item[Output:] hệ số Newton $c_k$; vector hệ số $b$ của $P_n$ theo biến $u$ ($u=x$ ở TH1, $u=t$ ở TH2); vector $a$ theo $x$; $v_j\approx f^{(j)}(\bar x)$.
\item[Bước 0 — Xử lý dữ liệu:] như câu 6 (chuẩn hoá, loại ô lỗi, gộp/báo mốc trùng, sắp tăng). Kiểm tra cách đều: $h:=\frac{x_n-x_0}{n}$, nếu $|x_{i+1}-x_i-h|\le10^{-6}h\ \forall i$ thì TH2, ngược lại TH1. $\bar x\notin[x_0,x_n]$ $\Rightarrow$ cảnh báo ngoại suy; $k$ nguyên $\ge0$.
\item[Bước 1 — Bảng:] $T_{i,0}:=y_i$; với $k=1..n$, $i=k..n$: TH1 $T_{i,k}:=\dfrac{T_{i,k-1}-T_{i-1,k-1}}{x_i-x_{i-k}}$; TH2 $T_{i,k}:=T_{i,k-1}-T_{i-1,k-1}$.
\item[Bước 2 — Hệ số:] tiến: $c_k:=T_{k,k}$; lùi: $c_k:=T_{n,k}$; TH2 chia thêm $k!$.
\item[Bước 3 — Nút tích:] TH1 tiến $z_i:=x_i$, lùi $z_i:=x_{n-i}$; TH2 tiến $z_i:=i$, lùi $z_i:=-i$ ($i=0..n-1$).
\item[Bước 4 — Bảng tích, cộng dồn:] $\omega:=(1)$, $b:=c_0$; với $k=1..n$: $\omega:=\omega\cdot(u-z_{k-1})$ (Horner nhân, câu 5a), $b:=b+c_k\,\omega$ (cộng theo bậc).
\item[Bước 5 — Đổi biến (TH2):] $a:=(b_n)$; với $j=n-1..0$: $a:=a\cdot(x-s)$, chia mọi hệ số cho $h$, $a_0:=a_0+b_j$ ($s=x_0$ tiến, $s=x_n$ lùi). TH1: $a:=b$.
\item[Bước 6 — Tại $\bar x$:] $\bar u:=\bar x$ (TH1) hoặc $\bar u:=(\bar x-s)/h$ (TH2); Horner chia lặp $b$ tại $\bar u$ (câu 4c) $\to d_j=P^{(j)}(\bar u)$; $v_j:=d_j$ (TH1) hoặc $d_j/h^j$ (TH2). Sai số $\le\frac{M_{n+1}}{(n+1)!}\prod|\bar x-x_i|$.
\item[Bước 7 — Kết thúc:] trả về $c$, $b$, $a$, $v$.
\end{thuattoan}

\section{Câu 15. Thuật toán Gauss I, Gauss II, Stirling, Bessel}

\pp{Bảng chọn ô.} Mốc cách đều, chỉ số tương đối với mốc trung tâm $x_0$, $t=\frac{x-x_0}{h}$, $P=\sum c_k\pi_k(t)$:

Công thức & Bộ mốc ($n+1$ mốc) & $c_k$ & $\pi_k(t)$, $\pi_0=1$\\\midrule
Gauss I & $x_{-\lfloor n/2\rfloor}..x_{\lceil n/2\rceil}$ & $\Delta^ky_{-\lfloor k/2\rfloor}/k!$ & $\pi_{k-1}(t-z_{k-1})$, $z=0,1,-1,2,\dots$\\

\section{Câu 18. Thuật toán cho bài toán nội suy ngược}

\pp{Bài toán.} Hàm $y=f(x)$ cho dạng bảng $(x_i,y_i)$, $i=0..n$; giải phương trình $f(x)=\bar y$.

\subsection{a) Tất cả các khoảng cách ly của $f(x)=\bar y$}
\begin{thuattoan}{Khoảng cách ly}
\item[Input:] bảng $(x_i,y_i)$ đã sắp tăng theo $x$; $\bar y$.
\item[Output:] danh sách nghiệm đúng tại mốc và danh sách khoảng cách ly $(x_i,x_{i+1})$.
\item[Bước 0 — Xử lý dữ liệu:] chuẩn hoá, loại ô lỗi, mốc trùng, sắp tăng (như câu 10). $\bar y\notin[\min y,\max y]\Rightarrow$ báo ``không có nghiệm trong bảng'', dừng.
\item[Bước 1 — Độ lệch:] $g_i:=y_i-\bar y$, $i=0..n$.
\item[Bước 2 — Với $i=0..n$:] 2.1: nếu $g_i=0$ thì $x_i$ là nghiệm; \ 2.2: nếu $i<n$ và $g_i\,g_{i+1}<0$ thì $(x_i,x_{i+1})$ là khoảng cách ly.
\item[Bước 3 — Kết thúc:] trả về hai danh sách (giả thiết lưới đủ mịn: mỗi khoảng chứa không quá một nghiệm).
\end{thuattoan}

\subsection{b) Các khoảng đơn điệu}
\begin{thuattoan}{Khoảng đơn điệu}
\item[Input:] bảng $(x_i,y_i)$, $n\ge1$. \quad\textbf{Output:} các đoạn $[x_a,x_b]$ kèm chiều tăng/giảm/hằng (hai đoạn kề chung một mốc — mốc cực trị).
\item[Bước 0 — Xử lý dữ liệu:] như câu a.
\item[Bước 1:] $s_i:=\operatorname{sign}(y_{i+1}-y_i)$, $i=0..n-1$; $a:=0$.
\item[Bước 2 — Với $i=1..n-1$:] nếu $s_i\ne s_{i-1}$: ghi đoạn $[x_a,x_i]$ chiều $s_{i-1}$; $a:=i$.
\item[Bước 3:] ghi đoạn cuối $[x_a,x_n]$ chiều $s_{n-1}$; trả về danh sách.
\end{thuattoan}

\subsection{c) Phương pháp hàm ngược}
\pp{Ý tưởng.} Trên đoạn mà $f$ đơn điệu chặt, tồn tại hàm ngược $x=f^{-1}(y)$; nội suy nó qua các mốc $(y_j,x_j)$ (mốc bất kỳ $\Rightarrow$ Newton tỷ sai phân) rồi thay $y=\bar y$. Sai số $\le\frac{\max|(f^{-1})^{(m)}|}{m!}\prod|\bar y-y_j|$ nên chọn các mốc có $|y_j-\bar y|$ nhỏ nhất.
\begin{thuattoan}{Hàm ngược}
\item[Input:] bảng; $\bar y$; một khoảng cách ly $(x_k,x_{k+1})$ nằm trong đoạn đơn điệu $[x_a,x_b]$ (câu a, b); số mốc $m\ge2$.
\item[Output:] nghiệm gần đúng $\bar x$ và ước lượng sai số.
\item[Bước 0 — Xử lý dữ liệu:] như câu a; $m$ nguyên $\ge2$.
\item[Bước 1 — Chọn mốc:] $L:=k$, $R:=k+1$; khi $R-L+1<m$: TH1 $L=a,R=b$: hết đoạn đơn điệu, dừng chọn (báo); TH2 $L=a$: $R:=R+1$; TH3 $R=b$: $L:=L-1$; TH4: kết nạp phía có $|y-\bar y|$ nhỏ hơn.
\item[Bước 2 — Đổi vai:] $u_s:=y_{L+s}$, $v_s:=x_{L+s}$; đoạn giảm thì đảo thứ tự để $u_0<u_1<\cdots$.
\item[Bước 3 — Tỷ sai phân:] bảng của $v$ theo $u$ (bước 1 câu 10, TH1); $c_j:=v[u_0..u_j]$.
\item[Bước 4 — Giá trị:] $\omega_0:=1$, $S:=c_0$; với $j\ge1$: $\omega_j:=\omega_{j-1}(\bar y-u_{j-1})$, $S:=S+c_j\omega_j$.
\item[Bước 5 — Kết thúc:] $\bar x:=S$; $|c_{\text{cuối}}\,\omega_{\text{cuối}}|$ là ước lượng sai số.
\end{thuattoan}

\subsection{d) Phương pháp lặp}
\pp{Ý tưởng.} Mốc cách đều, khoảng cách ly $(x_k,x_{k+1})$. Từ Newton tiến tại $x_k$: $\bar y=y_k+t\Delta y_k+\sum_{j\ge2}\frac{\Delta^jy_k}{j!}t(t-1)\cdots(t-j+1)$, giải theo $t$ ở số hạng bậc nhất:
\[ t=\varphi(t)=\frac{\bar y-y_k}{\Delta y_k}-\frac{1}{\Delta y_k}\sum_{j=2}^{m-1}\frac{\Delta^jy_k}{j!}t(t-1)\cdots(t-j+1),\qquad x=x_k+th. \]
Newton lùi tại $x_{k+1}$ (khi không đủ mốc bên phải): thay $\Delta^jy_k\to\nabla^jy_{k+1}$, $t(t-1)\cdots\to t(t+1)\cdots$, $x=x_{k+1}+th$. Lặp hội tụ khi $|\varphi'(t)|<1$ quanh nghiệm.
\begin{thuattoan}{Lặp}
\item[Input:] bảng cách đều bước $h$; $\bar y$; khoảng cách ly $(x_k,x_{k+1})$; số mốc $m$; $\varepsilon$; số lần lặp tối đa $N$.
\item[Output:] $\bar x$ và số lần lặp, hoặc thông báo không hội tụ.
\item[Bước 0 — Xử lý dữ liệu:] như câu a; kiểm tra cách đều (không $\Rightarrow$ dùng hàm ngược); $\varepsilon>0$, $N\ge1$ nguyên.
\item[Bước 1 — Chọn hướng:] TH1 $k+m-1\le n$: tiến, mốc $x_k..x_{k+m-1}$, $s:=k$; TH2 ngược lại mà $k+2-m\ge0$: lùi, mốc $x_{k+2-m}..x_{k+1}$, $s:=k+1$.
\item[Bước 2 — Sai phân:] bảng $T$ (bước 1 câu 10, TH2) trên $m$ mốc vừa chọn, đánh số lại $0..m-1$; TH1: $d_j:=T_{j,j}=\Delta^jy_k$ (đường chéo); TH2: $d_j:=T_{m-1,j}=\nabla^jy_{k+1}$ (hàng cuối).
\item[Bước 3 — Hàm lặp:] $\varphi(t)=A-\sum_{j=2}^{m-1}B_j\,\omega_j(t)$, $A=\frac{\bar y-d_0}{d_1}$, $B_j=\frac{d_j}{j!\,d_1}$, $\omega_j=t(t\mp1)\cdots(t\mp(j-1))$ (dấu $-$ TH1, $+$ TH2).
\item[Bước 4 — Khởi tạo:] $t_0:=A$.
\item[Bước 5 — Với $j=0..N-1$:] $t_{j+1}:=\varphi(t_j)$; nếu $h|t_{j+1}-t_j|<\varepsilon$: $\bar x:=x_s+t_{j+1}h$, trả về, dừng.
\item[Bước 6 — Không hội tụ:] hết $N$ lần $\Rightarrow$ báo ``Không hội tụ'' (in $t_N$ để tham khảo).
\end{thuattoan}

\section{Câu 20. Hàm ghép trơn (spline)}

\pp{Bài toán.} Cho $(x_i,y_i)$, $i=0..n$, $x_0<\dots<x_n$, $h_k=x_{k+1}-x_k$. Tìm $S(x)=S_k(x)$ trên $[x_k,x_{k+1}]$, $S_k$ đa thức bậc $p$, $S\in C^{p-1}$, $S(x_i)=y_i$. Đếm: $n(p+1)$ hệ số; nội suy $2n$ phương trình; trơn cấp $1..p-1$ tại $n-1$ mốc trong: $(p-1)(n-1)$; còn thiếu \textbf{$p-1$ điều kiện biên}.

\subsection{a) Xây dựng công thức}
\pp{Spline cấp 3.} Đặt $\sigma_k=S''(x_k)$. $S_k''$ bậc nhất nên $S_k''(x)=\dfrac{\sigma_k(x_{k+1}-x)+\sigma_{k+1}(x-x_k)}{h_k}$; tích phân hai lần:
\[ S_k(x)=\frac{\sigma_k(x_{k+1}-x)^3+\sigma_{k+1}(x-x_k)^3}{6h_k}+\alpha_k(x-x_k)+\beta_k(x_{k+1}-x). \]
$S_k(x_k)=y_k\Rightarrow\beta_k=\dfrac{y_k}{h_k}-\dfrac{\sigma_kh_k}{6}$; \ $S_k(x_{k+1})=y_{k+1}\Rightarrow\alpha_k=\dfrac{y_{k+1}}{h_k}-\dfrac{\sigma_{k+1}h_k}{6}$. Đặt $s_k=\dfrac{y_{k+1}-y_k}{h_k}$:
\[ S_k'(x)=\frac{-\sigma_k(x_{k+1}-x)^2+\sigma_{k+1}(x-x_k)^2}{2h_k}+s_k-\frac{(\sigma_{k+1}-\sigma_k)h_k}{6}, \]
\[ S_k'(x_{k+1})=s_k+\frac{h_k}{6}\sigma_k+\frac{h_k}{3}\sigma_{k+1},\qquad S_{k+1}'(x_{k+1})=s_{k+1}-\frac{h_{k+1}}{3}\sigma_{k+1}-\frac{h_{k+1}}{6}\sigma_{k+2}. \]
Trơn cấp 1 tại $x_{k+1}$ cho $n-1$ phương trình ($k=0..n-2$):
\[ \boxed{\frac{h_k}{6}\sigma_k+\frac{h_k+h_{k+1}}{3}\sigma_{k+1}+\frac{h_{k+1}}{6}\sigma_{k+2}=s_{k+1}-s_k} \]
cộng 2 điều kiện biên: $S''(x_0)=A\Rightarrow\sigma_0=A$; $S'(x_0)=A\Rightarrow\frac{h_0}{3}\sigma_0+\frac{h_0}{6}\sigma_1=s_0-A$; $S''(x_n)=B\Rightarrow\sigma_n=B$; $S'(x_n)=B\Rightarrow\frac{h_{n-1}}{6}\sigma_{n-1}+\frac{h_{n-1}}{3}\sigma_n=B-s_{n-1}$. Hệ ba đường chéo, chéo trội. Khai triển được dạng chính tắc $S_k=a_kx^3+b_kx^2+c_kx+d_k$:
\[ a_k=\frac{\sigma_{k+1}-\sigma_k}{6h_k},\ b_k=\frac{\sigma_kx_{k+1}-\sigma_{k+1}x_k}{2h_k},\ c_k=\frac{-\sigma_kx_{k+1}^2+\sigma_{k+1}x_k^2}{2h_k}+s_k-\frac{(\sigma_{k+1}-\sigma_k)h_k}{6}, \]
\[ d_k=\frac{\sigma_kx_{k+1}^3-\sigma_{k+1}x_k^3}{6h_k}+\frac{y_kx_{k+1}-y_{k+1}x_k}{h_k}+\frac{(\sigma_{k+1}x_k-\sigma_kx_{k+1})h_k}{6}. \]

\pp{Spline cấp 4.} Đặt $m_k=S'(x_k)$, $\sigma_k=S''(x_k)$, $M_k=S'''(x_k)$ ($S\in C^3$ nên các giá trị này chung cho hai mảnh kề). $S_k'''$ bậc nhất: $S_k'''(x)=M_k+\frac{M_{k+1}-M_k}{h_k}u$, $u=x-x_k$. Tích phân ba lần với giá trị ban đầu tại $x_k$ (khai triển Taylor):
\[ S_k(x)=y_k+m_ku+\frac{\sigma_k}{2}u^2+\frac{M_k}{6}u^3+\frac{M_{k+1}-M_k}{24h_k}u^4,\qquad u=x-x_k. \]
Liên tục của $S,S',S''$ tại $x_{k+1}$ ($S'''$ liên tục sẵn), $k=0..n-1$:
\[ \boxed{\begin{aligned}
&y_k+m_kh_k+\frac{\sigma_k}{2}h_k^2+\frac{3M_k+M_{k+1}}{24}h_k^3=y_{k+1},\\
&m_k+\sigma_kh_k+\frac{2M_k+M_{k+1}}{6}h_k^2=m_{k+1},\\
&\sigma_k+\frac{M_k+M_{k+1}}{2}h_k=\sigma_{k+1}.
\end{aligned}} \]
$3n$ phương trình, $3(n+1)$ ẩn $(m_k,\sigma_k,M_k)$ $\Rightarrow$ cần 3 điều kiện biên (vd $S'(x_0),S''(x_0),S'(x_n)$). Dạng chính tắc: khai triển $u=x-x_k$ bằng Horner nhân.

\noindent\textit{Đối chiếu:} cùng cách đó với $p=3$: $S_k=y_k+m_ku+\frac{\sigma_k}{2}u^2+\frac{\sigma_{k+1}-\sigma_k}{6h_k}u^3$, điều kiện $y_k+m_kh_k+\frac{2\sigma_k+\sigma_{k+1}}{6}h_k^2=y_{k+1}$, $m_{k+1}=m_k+\frac{\sigma_k+\sigma_{k+1}}{2}h_k$; khử $m_k$ được đúng hệ ba đường chéo ở trên.

\subsection{b) Thuật toán spline cấp $p\le4$}
\pp{Dạng tổng quát} (gộp các trường hợp trên): ẩn $D_{k,r}=S^{(r)}(x_k)$, $r=1..p-1$, $k=0..n$ ($D_{k,0}=y_k$);
\[ S_k(x)=\sum_{j=0}^{p-1}\frac{D_{k,j}}{j!}(x-x_k)^j+\frac{D_{k+1,p-1}-D_{k,p-1}}{p!\,h_k}(x-x_k)^p \]
($S^{(p-1)}$ tuyến tính từng khúc, tự liên tục). Điều kiện $S_k^{(r)}(x_{k+1})=D_{k+1,r}$, $r=0..p-2$:
\[ \sum_{j=\max(r,1)}^{p-1}\frac{h_k^{j-r}}{(j-r)!}D_{k,j}+\frac{h_k^{p-r-1}}{(p-r)!}(D_{k+1,p-1}-D_{k,p-1})-[r\ge1]D_{k+1,r}=[r=0](y_{k+1}-y_k). \]
\begin{thuattoan}{Spline cấp $p\le4$}
\item[Input:] $(x_i,y_i)$, $i=0..n$; bậc $p\in\{1,2,3,4\}$; $p-1$ điều kiện biên $S^{(r)}(\xi)=v$ với $\xi\in\{x_0,x_n\}$, $1\le r\le p-1$.
\item[Output:] với mỗi $k$: hệ số chính tắc của $S_k$ (từ bậc cao xuống).
\item[Bước 0 — Xử lý dữ liệu:] chuẩn hoá số, loại ô lỗi; mốc trùng khác $y$ $\Rightarrow$ dừng (spline phải qua mọi điểm); \textbf{sắp tăng theo $x$}; $n\ge1$; $p$ nguyên $1..4$; điều kiện biên: đọc dạng \texttt{S'(0)=0}, điểm phải là $x_0$ hoặc $x_n$, cấp $1..p-1$, không lặp, đúng $p-1$ điều kiện (sai $\Rightarrow$ dừng, báo rõ).
\item[Bước 1 — Chọn cách giải:] $p=1$: $S_k=\frac{y_{k+1}-y_k}{h_k}x+\frac{y_kx_{k+1}-y_{k+1}x_k}{h_k}$, sang bước 5. $p=2$ (cho $S'(x_0)$ hoặc $S'(x_n)$): bước 2. $p=3$, mỗi đầu một điều kiện $S'$/$S''$: bước 3. Còn lại: bước 4.
\item[Bước 2 — Cấp 2:] $m_k=S'(x_k)$: $\gamma_k:=\frac{2(y_{k+1}-y_k)}{h_k}$; cho $m_0$: $m_{k+1}:=\gamma_k-m_k$ ($k=0..n-1$); cho $m_n$: $m_k:=\gamma_k-m_{k+1}$ ($k=n-1..0$). Hệ số: $a_k=\frac{m_{k+1}-m_k}{2h_k}$, $b_k=\frac{m_kx_{k+1}-m_{k+1}x_k}{h_k}$, $c_k=\frac{-m_kx_{k+1}^2+m_{k+1}x_k^2}{2h_k}+y_k+\frac{m_kh_k}{2}$. Sang bước 5.
\item[Bước 3 — Cấp 3:] lập hệ ba đường chéo $\ell_i\sigma_{i-1}+d_i\sigma_i+u_i\sigma_{i+1}=r_i$ ($i=0..n$, hàng trong và hàng biên như câu a); giải bằng truy đuổi: $P_0=-u_0/d_0$, $Q_0=r_0/d_0$; $\mu=d_i+\ell_iP_{i-1}$ ($\mu=0\Rightarrow$ suy biến, dừng), $P_i=-u_i/\mu$, $Q_i=(r_i-\ell_iQ_{i-1})/\mu$; $\sigma_n=Q_n$, $\sigma_i=P_i\sigma_{i+1}+Q_i$. Hệ số $a_k,b_k,c_k,d_k$ theo câu a. Sang bước 5.
\item[Bước 4 — Tổng quát:] lập hệ $(p-1)(n+1)$ phương trình (điều kiện tại $x_{k+1}$ ở trên và $D_{k,r}=v$ ở biên); khử Gauss chọn trụ (trụ $\approx0\Rightarrow$ điều kiện biên không xác định duy nhất spline, dừng); hệ số địa phương $e_{k,j}=D_{k,j}/j!$, $e_{k,p}=\frac{D_{k+1,p-1}-D_{k,p-1}}{p!h_k}$; khai triển $u=x-x_k$ (Horner nhân) ra dạng chính tắc.
\item[Bước 5 — Kết thúc:] trả về các $S_k$.
\end{thuattoan}

\section{Câu 22. Thuật toán phương pháp bình phương tối thiểu}

\subsection{a) Hàm thực nghiệm với tập cơ sở $\{\varphi_j\}_{j=1}^m$ và sai số trung bình phương}
\pp{Cơ sở.} Tìm $g=\sum_{j=1}^ma_j\varphi_j$ cực tiểu $S(a)=\sum_{i=1}^n\big(g(x_i)-y_i\big)^2$. Với $\Phi=(\varphi_j(x_i))_{n\times m}$: $S=a^tMa-2y^t\Phi a+y^ty$, $M=\Phi^t\Phi$; $\nabla S=0\Leftrightarrow Ma=\Phi^ty$ (hệ chuẩn; $M$ đối xứng, xác định dương khi các cột $\Phi$ độc lập). Sai số trung bình phương $\sigma=\sqrt{\frac1n\sum(g(x_i)-y_i)^2}$.
\begin{thuattoan}{Bình phương tối thiểu tuyến tính theo tham số}
\item[Input:] $(x_i,y_i)$, $i=1..n$; cơ sở $\varphi_1..\varphi_m$ (biểu thức theo $x$).
\item[Output:] $a=(a_1..a_m)$; $\sigma$.
\item[Bước 0 — Xử lý dữ liệu:] đọc cặp $(x_i,y_i)$ (file nhiều bộ dữ liệu cạnh nhau thì chọn bộ); chuẩn hoá số, loại ô lỗi và báo; \emph{không} gộp mốc trùng (cho phép nhiều quan sát tại một $x$), không cần sắp xếp; loại điểm làm $\varphi_j(x_i)$ không xác định (vd $1/x$ tại $x=0$); cần $n\ge m$.
\item[Bước 1 — Ma trận cơ sở:] $\Phi_{ij}:=\varphi_j(x_i)$.
\item[Bước 2 — Hệ chuẩn:] $M_{jk}:=\sum_i\Phi_{ij}\Phi_{ik}$, $b_j:=\sum_i\Phi_{ij}y_i$.
\item[Bước 3 — Giải $Ma=b$:] khử Gauss chọn trụ: với $c=1..m$: chọn hàng trụ $|M_{pc}|$ lớn nhất ($\approx0\Rightarrow$ ``ma trận suy biến: cơ sở phụ thuộc tuyến tính trên dữ liệu'', dừng), đổi hàng, khử phía dưới; thế ngược.
\item[Bước 4 — Sai số:] $g_i:=\sum_ja_j\Phi_{ij}$; $\sigma:=\sqrt{\frac1n\sum_i(g_i-y_i)^2}$.
\item[Bước 5 — Kết thúc:] trả về $a$, $\sigma$.
\end{thuattoan}

\subsection{b) Dạng $y=a\,e^{b_1\varphi_1(x)+b_2\varphi_2(x)}$ và $y=a\,x^b$}
\pp{Tuyến tính hoá.} $y_i$ cùng dấu $s$: $\ln(sy)=\ln(sa)+b_1\varphi_1+b_2\varphi_2$ — tuyến tính theo $(\ln(sa),b_1,b_2)$ với cơ sở $(1,\varphi_1,\varphi_2)$ ($\varphi_j$ không được là hằng, vì hằng trùng vai trò với $\ln a$). $y=ax^b$: $\ln(sy)=\ln(sa)+b\ln x$ (dạng trên với $\varphi_1=\ln x$, cần $x_i>0$).

\pp{Lưu ý.} Bình phương tối thiểu trên $\ln y$ cực tiểu $\sum(\ln y_i-\ln g_i)^2\approx\sum\big(\frac{y_i-g_i}{y_i}\big)^2$ — sai số \emph{tương đối}, nên điểm có $|y_i|$ nhỏ (cỡ nhiễu) bị khuếch đại. Vì vậy thêm bước hiệu chỉnh Gauss--Newton để cực tiểu đúng $\sum(y_i-g_i)^2$; mỗi bước lại là một bài toán bình phương tối thiểu tuyến tính có trọng số.
\begin{thuattoan}{Hàm dạng mũ / luỹ thừa}
\item[Input:] $(x_i,y_i)$; cơ sở $\varphi_1,\varphi_2$ (luỹ thừa: $\varphi_1=\ln x$); $\varepsilon$, $N$ cho bước hiệu chỉnh.
\item[Output:] $a$, $b_j$, $\sigma$ (trên $y$ gốc).
\item[Bước 0 — Xử lý dữ liệu:] như câu a; thêm: $y_i=0$ hoặc trái dấu đa số $\Rightarrow$ loại điểm và báo (không lấy $\ln$ được), loại quá nửa $\Rightarrow$ dừng; dạng luỹ thừa loại $x_i\le0$; $\varphi_j$ là hằng trên lưới $\Rightarrow$ dừng.
\item[Bước 1 — Dấu:] $s:=\operatorname{sign}y_i$.
\item[Bước 2 — Đổi biến:] $z_i:=\ln(sy_i)$; cơ sở $(1,\varphi_1,\varphi_2)$.
\item[Bước 3 — Bình phương tối thiểu:] $(c_0,c_1,c_2)$ theo câu a với dữ liệu $(x_i,z_i)$.
\item[Bước 4 — Trả biến:] $a:=s\,e^{c_0}$, $b_j:=c_j$; $\sigma:=\sqrt{\frac1n\sum(a\,e^{\sum b_j\varphi_j(x_i)}-y_i)^2}$.
\item[Bước 5 — Hiệu chỉnh (Gauss--Newton):] khởi tạo $c$ bằng bình phương tối thiểu có trọng số $w_i=y_i^2$ trên $z_i$; lặp $j=0..N-1$: $g_i:=s\,e^{(\Phi c)_i}$, $\tilde z_i:=\ln(sg_i)+\frac{y_i-g_i}{g_i}$, $w_i:=g_i^2$, $c^{\text{mới}}$ := nghiệm của $\Phi^tW\Phi c=\Phi^tW\tilde z$; dừng khi $\max|c^{\text{mới}}-c|<\varepsilon(1+\max|c|)$; hết $N$ lần $\Rightarrow$ báo không hội tụ, giữ kết quả bước 4.
\item[Bước 6 — Kết thúc:] chọn kết quả có $\sigma$ nhỏ hơn; trả về $a$, $b$, $\sigma$.
\end{thuattoan}
