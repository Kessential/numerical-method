\section{Câu 3. Mốc nội suy tối ưu}

\subsection{a) Xây dựng công thức $n$ mốc nội suy tối ưu trên $[-1,1]$ và trên $[a,b]$}

\pp{Sai số nội suy.} Với $n$ mốc $x_0,\dots,x_{n-1}\in[a,b]$ và $P$ bậc $\le n-1$:
\[ f(x)-P(x)=\frac{f^{(n)}(\xi)}{n!}\,w_n(x),\qquad w_n(x)=\prod_{i=0}^{n-1}(x-x_i). \]
$w_n$ là đa thức bậc $n$ có hệ số cao nhất bằng 1; mốc tối ưu là mốc làm $\max_{[a,b]}|w_n(x)|$ nhỏ nhất.

\pp{Trên $[-1,1]$.} Đa thức Chebyshev $T_n(t)=\cos(n\arccos t)$ có hệ số cao nhất $2^{n-1}$ và $|T_n(t)|\le1$. Trong các đa thức bậc $n$ có hệ số cao nhất bằng 1, $\tilde T_n=2^{1-n}T_n$ có $\max_{[-1,1]}|\tilde T_n|=2^{1-n}$ nhỏ nhất. Vậy chọn $w_n=\tilde T_n$, tức các mốc là nghiệm của $T_n$:
\[ \boxed{t_i=\cos\frac{(2i+1)\pi}{2n},\ i=0,\dots,n-1,\qquad \min\max_{[-1,1]}|w_n|=\frac{1}{2^{n-1}}.} \]

\pp{Trên $[a,b]$.} Đổi biến $x=\dfrac{a+b}{2}+\dfrac{b-a}{2}t$ ánh xạ $[-1,1]$ lên $[a,b]$. Khi đó $x-x_i=\frac{b-a}{2}(t-t_i)$ nên $w_n(x)=\left(\frac{b-a}{2}\right)^n\prod(t-t_i)$; bài toán trên $[a,b]$ tương đương bài toán trên $[-1,1]$:
\[ \boxed{x_i=\frac{a+b}{2}+\frac{b-a}{2}\cos\frac{(2i+1)\pi}{2n},\qquad \min\max_{[a,b]}|w_n|=\left(\frac{b-a}{2}\right)^n\frac{1}{2^{n-1}}=\frac{(b-a)^n}{2^{2n-1}},} \]
và khi đó $|f(x)-P(x)|\le\dfrac{M_n}{n!}\cdot\dfrac{(b-a)^n}{2^{2n-1}}$, $M_n=\max_{[a,b]}|f^{(n)}|$.

\pp{Sắp tăng dần.} $\cos$ giảm trên $[0,\pi]$ nên $t_i$ ở trên giảm theo $i$. Dùng $t_i=-\cos\frac{(2i+1)\pi}{2n}=\cos\frac{(2(n-1-i)+1)\pi}{2n}$ (cùng tập nghiệm, đánh số ngược) thì $t_0<t_1<\dots<t_{n-1}$.

\subsection{b) Thuật toán xác định $n$ mốc tối ưu sắp tăng dần trên $[a,b]$}
\begin{thuattoan}{Mốc nội suy tối ưu}
\item[Input:] số mốc $n$; đoạn $[a,b]$.
\item[Output:] mảng $x=(x_0,\dots,x_{n-1})$, $x_0<x_1<\dots<x_{n-1}$.
\item[Bước 0:] \textbf{Xử lý dữ liệu:} Nếu $n<1$ thì báo lỗi, dừng thuật toán.\newline Nếu $a=b$ thì báo lỗi, dừng thuật toán.\newline Nếu $a>b$ thì đổi chỗ $a$ và $b$.
\item[Bước 1:] \textbf{Tính các mốc:} Lặp $i=0,\dots,n-1$: tính $x_i:=\dfrac{a+b}{2}-\dfrac{b-a}{2}\cos\dfrac{(2i+1)\pi}{2n}$.
\item[Bước 2:] \textbf{Kết thúc:} Dừng thuật toán, trả về $x$.
\end{thuattoan}

\subsection{c) Áp dụng}
\pp{Phương pháp.} Mốc tối ưu là nghiệm của đa thức Chebyshev $T_n$ đổi biến từ $[-1,1]$ sang $[a,b]$, sắp tăng dần (câu a, b): $x_i=\dfrac{a+b}{2}-\dfrac{b-a}{2}\cos\dfrac{(2i+1)\pi}{2n}$, $i=0,\dots,n-1$.

\section{Câu 4. Sơ đồ Horner: giá trị, phép chia, đạo hàm}

\subsection{a) Giá trị $P_n(c)$ \quad b) Chia $P_n(x)$ cho $(x-c)$}
\begin{thuattoan}{Horner chia (giá trị, thương, số dư)}
\item[Input:] vector hệ số $a=(a_n,\dots,a_0)$; số $c$.
\item[Output:] thương $b=(b_{n-1},\dots,b_0)$; số dư $r=P_n(c)$.
\item[Bước 0:] \textbf{Xử lý dữ liệu:} Bỏ các hệ số $0$ ở đầu vector $a$.\newline Nếu không còn hệ số nào ($P_n\equiv0$) thì dừng thuật toán, trả về $b:=(0)$ và $r:=0$.\newline Đặt $n:=(\text{số phần tử của }a)-1$.
\item[Bước 1:] \textbf{Đa thức hằng:} Nếu $n=0$ thì dừng thuật toán, trả về $b:=(0)$ và $r:=a_0$.
\item[Bước 2:] \textbf{Khởi tạo:} Đặt $b_{n-1}:=a_n$.
\item[Bước 3:] \textbf{Lặp chia:} Lặp $i=n-1,\dots,1$: tính $b_{i-1}:=a_i+c\,b_i$.
\item[Bước 4:] \textbf{Số dư:} Tính $r:=a_0+c\,b_0$.\newline Dừng thuật toán, trả về $b$ và $r$.
\end{thuattoan}

\subsection{c) Đạo hàm cấp $k$ của $P_n$ tại $c$}
\begin{thuattoan}{Đạo hàm cấp $0..k$ tại $c$ (Horner chia lặp)}
\item[Input:] vector hệ số $a=(a_n,\dots,a_0)$; số $c$; cấp $k$.
\item[Output:] vector $r=(r_0,\dots,r_K)$, $K=\min(k,n)$; vector $v=(v_0,\dots,v_k)$ với $v_j=P_n^{(j)}(c)$.
\item[Bước 0:] \textbf{Xử lý dữ liệu:} Nếu $k<0$ thì báo lỗi, dừng thuật toán.\newline Bỏ các hệ số $0$ ở đầu vector $a$.\newline Nếu không còn hệ số nào ($P_n\equiv0$) thì dừng thuật toán, trả về $r:=(0)$ và $v:=(0,\dots,0)$ ($k+1$ phần tử).\newline Đặt $n:=(\text{số phần tử của }a)-1$.
\item[Bước 1:] \textbf{Khởi tạo:} Đặt $K:=\min(k,n)$.\newline Lặp $i=0,\dots,n$: đặt $b_i:=a_i$.
\item[Bước 2:] \textbf{Chia lặp:} Lặp $j=0,\dots,K$:
  \begin{con}
  \item[2.1:] Lặp $i=n-1,\dots,j$: tính $b_i:=b_i+c\,b_{i+1}$ (chia cho $x-c$).
  \item[2.2:] Đặt $r_j:=b_j$. Tính $v_j:=j!\,r_j$.
  \end{con}
\item[Bước 3:] \textbf{Đạo hàm bậc cao:} Lặp $j=K+1,\dots,k$: đặt $v_j:=0$.\newline Dừng thuật toán, trả về $r$ và $v$.
\end{thuattoan}

\subsection{d) Áp dụng: $P(x)=<<c4_P>>$, $a=(<<c4_a>>)$}
\pp{Phương pháp} (Horner chia lặp, câu c). Chia liên tiếp $P$ cho $(x-c)$; số dư lần chia thứ $j$ là $r_j=\frac{P^{(j)}(c)}{j!}$. Hàng ``Hệ số'' là $b^{(-1)}_i=a_i$; hàng $j$ ghi $b^{(j)}_i$ ($i=<<c4_n>>,\dots,j$) sau lần chia thứ $j$, hàng $c$ phía trên ghi $c\,b^{(j)}_{i+1}$:
\[ b^{(j)}_<<c4_ni>>=a_<<c4_ni>>,\qquad b^{(j)}_i=b^{(j-1)}_i+c\,b^{(j)}_{i+1}\ \ (i=<<c4_n1>>,\dots,j),\qquad r_j=b^{(j)}_j,\qquad P^{(j)}(c)=j!\,r_j,\qquad j=0,\dots,<<c4_K>>. \]

\section{Câu 10. Thuật toán đa thức nội suy Newton tiến / lùi}

\item[Bước 0:] \textbf{Xử lý dữ liệu:} Nếu $x$, $y$ khác số phần tử thì báo lỗi, dừng thuật toán.\newline Sắp các mốc tăng theo $x$.\newline Nếu hai mốc trùng nhau thì: cùng $y$ giữ một mốc, khác $y$ báo lỗi, dừng thuật toán.\newline Đặt $n:=(\text{số mốc})-1$.\newline Nếu $r<0$ thì báo lỗi, dừng thuật toán.

\item[Bước 2:] \textbf{Hệ số:} Lặp $k=0,\dots,n$:\thut nếu tiến thì đặt $c_k:=T_{k,k}$;\thut nếu lùi thì đặt $c_k:=T_{n,k}$.
\item[Bước 3:] \textbf{Bảng tích, cộng dồn:} Khởi tạo vector $e=(e_0,e_1,\dots,e_n):=(1,0,\dots,0)$, $a=(a_0,a_1,\dots,a_n):=(c_0,0,\dots,0)$, gồm $n+1$ phần tử.\newline Lặp $k=1,\dots,n$:
  \begin{con}
  \item[3.1:] Nếu tiến thì đặt $z:=x_{k-1}$; nếu lùi thì đặt $z:=x_{n-k+1}$.\newline Nhân vector $e$ với $(x-z)$ theo thuật toán Câu 5a.
  \item[3.2:] Lặp $l=0,\dots,k$: tính $a_l:=a_l+c_k\,e_l$.
  \end{con}
\item[Bước 4:] \textbf{Giá trị, đạo hàm:} Gọi thuật toán Câu 4c với vector $a$, $c:=\bar x$, cấp $k:=r$, được vector $f=(f_0,\dots,f_r)$, $f_j=P_n^{(j)}(\bar x)$.\newline Dừng thuật toán, trả về $a$ và $f$.
\end{thuattoan}

\subsection{a) Gauss I, Gauss II}
\begin{thuattoan}{Gauss I / Gauss II}
\item[Input:] vector $x$, $y$ (mốc cách đều); điểm $\bar x$; cấp đạo hàm $r$; số mốc dùng $m$; công thức (Gauss I hoặc Gauss II).

\item[Bước 1:] \textbf{Trích xuất, mốc trung tâm:} Gọi thuật toán Câu 11a với $x$, $y$, $\bar x$, $k:=m$, được $X$, $Y$.\newline Đặt $n:=m-1$.\newline Nếu Gauss I thì đặt $q:=\lfloor n/2\rfloor$; nếu Gauss II thì đặt $q:=\lceil n/2\rceil$.\newline Lấy mốc thứ $q$ của $X$ (đếm từ $0$) làm $x_0$, đánh số lại $X$, $Y$ thành $x_i$, $y_i$, $i=-q,\dots,n-q$.

  \item[4.1:] Nhân vector $e$ với $(t-z_{k-1})$ theo thuật toán Câu 5a.
  \item[4.2:] Lặp $l=0,\dots,k$: tính $b_l:=b_l+c_k\,e_l$.

\item[Output:] vector $b=(b_n,\dots,b_0)$ với $P(t)=\sum_lb_lt^l$, $t=\frac{x-x_0}{h}$; vector $f=(f_0,\dots,f_r)$, $f_j\approx f^{(j)}(\bar x)$.

\section{Câu 18. Thuật toán cho bài toán nội suy ngược}

\subsection{a) Tất cả các khoảng cách ly của $f(x)=\bar y$}
\begin{thuattoan}{Khoảng cách ly}
\item[Input:] vector $x$, $y$ ($n+1$ mốc); giá trị $\bar y$.
\item[Output:] danh sách $N$ các mốc là nghiệm; danh sách $K$ các khoảng cách ly $(x_i,x_{i+1})$.
\item[Bước 0:] \textbf{Xử lý dữ liệu:} Nếu $x$, $y$ khác số phần tử thì báo lỗi, dừng thuật toán.\newline Sắp các mốc tăng theo $x$.\newline Nếu hai mốc trùng nhau thì: cùng $y$ giữ một mốc, khác $y$ báo lỗi, dừng thuật toán.\newline Đặt $n:=(\text{số mốc})-1$.
\item[Bước 1:] \textbf{Khởi tạo:} Đặt $N:=\varnothing$, $K:=\varnothing$.\newline Lặp $i=0,\dots,n$: tính $g_i:=y_i-\bar y$.
\item[Bước 2:] \textbf{Tìm khoảng cách ly:} Lặp $i=0,\dots,n$:\thut nếu $g_i=0$ thì thêm $x_i$ vào $N$;\thut nếu $i<n$ và $g_i\,g_{i+1}<0$ thì thêm $(x_i,x_{i+1})$ vào $K$.
\item[Bước 3:] \textbf{Kết thúc:} Nếu $N$ và $K$ đều rỗng thì báo ``phương trình không có nghiệm trong $[x_0,x_n]$''.\newline Dừng thuật toán, trả về $N$ và $K$.
\end{thuattoan}

\subsection{b) Các khoảng đơn điệu}
\begin{thuattoan}{Khoảng đơn điệu}
\item[Input:] vector $x$, $y$ ($n+1$ mốc).
\item[Output:] danh sách các đoạn $[x_a,x_b]$ kèm chiều tăng / giảm / hằng.
\item[Bước 0:] \textbf{Xử lý dữ liệu:} Nếu $x$, $y$ khác số phần tử thì báo lỗi, dừng thuật toán.\newline Sắp các mốc tăng theo $x$.\newline Nếu hai mốc trùng nhau thì: cùng $y$ giữ một mốc, khác $y$ báo lỗi, dừng thuật toán.\newline Đặt $n:=(\text{số mốc})-1$.\newline Nếu $n<1$ thì báo lỗi (cần ít nhất 2 mốc), dừng thuật toán.
\item[Bước 1:] \textbf{Dấu của $\Delta y_i$:} Lặp $i=0,\dots,n-1$: tính $s_i:=\operatorname{sign}(y_{i+1}-y_i)$.\newline Đặt $a:=0$.
\item[Bước 2:] \textbf{Tách đoạn:} Lặp $i=1,\dots,n-1$:\thut nếu $s_i\ne s_{i-1}$ thì ghi đoạn $[x_a,x_i]$ với chiều $s_{i-1}$ và đặt $a:=i$.
\item[Bước 3:] \textbf{Đoạn cuối:} Ghi đoạn $[x_a,x_n]$ với chiều $s_{n-1}$.\newline Dừng thuật toán, trả về danh sách các đoạn.
\end{thuattoan}

\subsection{c) Phương pháp hàm ngược}
\begin{thuattoan}{Hàm ngược}
\item[Input:] vector $x$, $y$; $\bar y$; khoảng cách ly $(x_k,x_{k+1})$ nằm trong đoạn đơn điệu $[x_a,x_b]$ (câu a, b); số mốc $m$.
\item[Output:] nghiệm gần đúng $\bar x$.
\item[Bước 0:] \textbf{Xử lý dữ liệu:} Nếu $x$, $y$ khác số phần tử thì báo lỗi, dừng thuật toán.\newline Sắp các mốc tăng theo $x$.\newline Nếu hai mốc trùng nhau thì: cùng $y$ giữ một mốc, khác $y$ báo lỗi, dừng thuật toán.\newline Đặt $n:=(\text{số mốc})-1$.\newline Nếu $m<2$ thì báo lỗi, dừng thuật toán.
\item[Bước 1:] \textbf{Khởi tạo:} Đặt $L:=k$, $R:=k+1$.
\item[Bước 2:] \textbf{Chọn mốc:} Lặp $s=3,\dots,m$:\thut nếu $L=a$ và $R=b$ thì dừng lặp;\thut ngược lại nếu $L=a$ thì $R:=R+1$;\thut ngược lại nếu $R=b$ thì $L:=L-1$;\thut ngược lại nếu $|y_{L-1}-\bar y|\le|y_{R+1}-\bar y|$ thì $L:=L-1$, còn không thì $R:=R+1$.
\item[Bước 3:] \textbf{Đổi vai trò $x$, $y$:} Đặt $u:=(y_L,\dots,y_R)$, $v:=(x_L,\dots,x_R)$ (đánh số lại $0,\dots,R-L$).\newline Nếu đoạn giảm thì đảo thứ tự $u$, $v$ để $u_0<u_1<\cdots$.
\item[Bước 4:] \textbf{Bảng tỷ sai phân:} Khởi tạo ma trận $T$ cỡ $(R-L+1)\times(R-L+1)$ (tỷ sai phân của $v$ theo $u$).\newline Lặp $i=0,\dots,R-L$: đặt $T_{i,0}:=v_i$.\newline Lặp lồng $j=1,\dots,R-L$; $i=j,\dots,R-L$: tính $T_{i,j}:=\dfrac{T_{i,j-1}-T_{i-1,j-1}}{u_i-u_{i-j}}$.\newline Lặp $j=0,\dots,R-L$: đặt $c_j:=T_{j,j}$.
\item[Bước 5:] \textbf{Giá trị tại $\bar y$:} Đặt $\omega:=1$, $S:=c_0$.\newline Lặp $j=1,\dots,R-L$:\thut tính $\omega:=\omega\,(\bar y-u_{j-1})$;\thut tính $S:=S+c_j\,\omega$.
\item[Bước 6:] \textbf{Kết thúc:} Đặt $\bar x:=S$.\newline Dừng thuật toán, trả về $\bar x$ (số hạng cuối $|c_{R-L}\,\omega|$ là ước lượng sai số).
\end{thuattoan}

\subsection{d) Phương pháp lặp}
\begin{thuattoan}{Lặp Newton tiến / lùi giải $f(x)=\bar y$}
\item[Input:] vector $x$, $y$ ($n+1$ mốc cách đều bước $h$); $\bar y$; khoảng cách ly $(x_k,x_{k+1})$; số mốc $m$; sai số $\varepsilon$; số lần lặp tối đa $N$.
\item[Output:] nghiệm gần đúng $\bar x$, hoặc thông báo ``Không hội tụ''.
\item[Bước 0:] \textbf{Xử lý dữ liệu:} Nếu $x$, $y$ khác số phần tử thì báo lỗi, dừng thuật toán.\newline Sắp các mốc tăng theo $x$.\newline Nếu hai mốc trùng nhau thì: cùng $y$ giữ một mốc, khác $y$ báo lỗi, dừng thuật toán.\newline Đặt $n:=(\text{số mốc})-1$.\newline Nếu $n<1$ thì báo lỗi (cần ít nhất 2 mốc), dừng thuật toán.\newline Đặt $h:=\dfrac{x_n-x_0}{n}$.\newline Lặp $i=0,\dots,n-1$:\thut nếu $|x_{i+1}-x_i-h|>10^{-6}h$ thì tìm đoạn đơn điệu $[x_a,x_b]$ chứa $(x_k,x_{k+1})$ theo thuật toán câu b, gọi thuật toán câu c với $x$, $y$, $\bar y$, $k$, $a$, $b$, $m$ được $\bar x$, dừng thuật toán, trả về $\bar x$.\newline Nếu $m<2$, $\varepsilon\le0$ hoặc $N<1$ thì báo lỗi, dừng thuật toán.\newline Nếu $k+m-1>n$ và $k+2-m<0$ thì đặt $m:=\max(n-k+1,\,k+2)$.
\item[Bước 1:] \textbf{Chọn hướng và mốc gốc:}\thut nếu $k+m-1\le n$ thì lặp tiến, đặt $s:=k$;\thut trái lại lặp lùi, đặt $s:=k+1$.
\item[Bước 2:] \textbf{Bảng sai phân:} Khởi tạo ma trận $T$ cỡ $m\times m$ (bảng sai phân của $m$ mốc, đánh số lại $0,\dots,m-1$).\newline Lặp $i=0,\dots,m-1$:\thut nếu tiến thì đặt $T_{i,0}:=y_{k+i}$;\thut nếu lùi thì đặt $T_{i,0}:=y_{k+2-m+i}$.\newline Lặp lồng $j=1,\dots,m-1$; $i=j,\dots,m-1$: tính $T_{i,j}:=T_{i,j-1}-T_{i-1,j-1}$.\newline Lặp $j=0,\dots,m-1$:\thut nếu tiến thì đặt $d_j:=T_{j,j}$ ($=\Delta^jy_k$);\thut nếu lùi thì đặt $d_j:=T_{m-1,j}$ ($=\nabla^jy_{k+1}$).
\item[Bước 3:] \textbf{Hàm lặp:} Nếu tiến thì $\varphi(t):=\dfrac{\bar y-d_0}{d_1}-\sum_{j=2}^{m-1}\dfrac{d_j}{j!\,d_1}\,t(t-1)\cdots(t-j+1)$.\newline Nếu lùi thì $\varphi(t):=\dfrac{\bar y-d_0}{d_1}-\sum_{j=2}^{m-1}\dfrac{d_j}{j!\,d_1}\,t(t+1)\cdots(t+j-1)$.
\item[Bước 4:] \textbf{Khởi tạo:} Đặt $t_0:=\dfrac{\bar y-d_0}{d_1}$, $l:=0$.
\item[Bước 5:] \textbf{Lặp tìm $t$:} Tính $t_{l+1}:=\varphi(t_l)$.
\item[Bước 6:] \textbf{Kiểm tra dừng:} Nếu $h\,|t_{l+1}-t_l|<\varepsilon$ thì:\thut nếu (tiến và $0\le t_{l+1}\le1$) hoặc (lùi và $-1\le t_{l+1}\le0$) thì đặt $\bar x:=x_s+t_{l+1}h$, dừng thuật toán, trả về $\bar x$;\thut ngược lại báo ``nghiệm lặp nằm ngoài khoảng cách ly'', dừng thuật toán.
\item[Bước 7:] \textbf{Không hội tụ:} Nếu $l+1=N$ thì báo ``Không hội tụ'', dừng thuật toán.\newline Trái lại đặt $l:=l+1$, quay lại bước 5.
\end{thuattoan}

\section{Câu 20. Hàm ghép trơn (spline)}

\subsection{a) Xây dựng công thức}
Spline cấp $p$: $S(x)=S_k(x)$ trên $[x_k,x_{k+1}]$, $S_k$ đa thức bậc $p$, $S\in C^{p-1}[x_0,x_n]$, $S(x_i)=y_i$; $h_k=x_{k+1}-x_k$; cần thêm $p-1$ điều kiện biên.

\pp{Spline cấp 3.} Đặt $\sigma_k=S''(x_k)$. $S_k''$ bậc nhất nên $S_k''(x)=\dfrac{\sigma_k(x_{k+1}-x)+\sigma_{k+1}(x-x_k)}{h_k}$; tích phân hai lần:
\[ S_k(x)=\frac{\sigma_k(x_{k+1}-x)^3+\sigma_{k+1}(x-x_k)^3}{6h_k}+\alpha_k(x-x_k)+\beta_k(x_{k+1}-x). \]
$S_k(x_k)=y_k\Rightarrow\beta_k=\dfrac{y_k}{h_k}-\dfrac{\sigma_kh_k}{6}$; \ $S_k(x_{k+1})=y_{k+1}\Rightarrow\alpha_k=\dfrac{y_{k+1}}{h_k}-\dfrac{\sigma_{k+1}h_k}{6}$. Đặt $s_k=\dfrac{y_{k+1}-y_k}{h_k}$:
\[ S_k'(x)=\frac{-\sigma_k(x_{k+1}-x)^2+\sigma_{k+1}(x-x_k)^2}{2h_k}+s_k-\frac{(\sigma_{k+1}-\sigma_k)h_k}{6}, \]
\[ S_k'(x_{k+1})=s_k+\frac{h_k}{6}\sigma_k+\frac{h_k}{3}\sigma_{k+1},\qquad S_{k+1}'(x_{k+1})=s_{k+1}-\frac{h_{k+1}}{3}\sigma_{k+1}-\frac{h_{k+1}}{6}\sigma_{k+2}. \]
Trơn cấp 1 tại $x_{k+1}$ cho $n-1$ phương trình ($k=0..n-2$):
\[ \boxed{\frac{h_k}{6}\sigma_k+\frac{h_k+h_{k+1}}{3}\sigma_{k+1}+\frac{h_{k+1}}{6}\sigma_{k+2}=s_{k+1}-s_k} \]
cộng 2 điều kiện biên: $S''(x_0)=A\Rightarrow\sigma_0=A$; $S'(x_0)=A\Rightarrow\frac{h_0}{3}\sigma_0+\frac{h_0}{6}\sigma_1=s_0-A$; $S''(x_n)=B\Rightarrow\sigma_n=B$; $S'(x_n)=B\Rightarrow\frac{h_{n-1}}{6}\sigma_{n-1}+\frac{h_{n-1}}{3}\sigma_n=B-s_{n-1}$. Hệ ba đường chéo, chéo trội. Khai triển được dạng chính tắc $S_k=a_kx^3+b_kx^2+c_kx+d_k$:
\[ a_k=\frac{\sigma_{k+1}-\sigma_k}{6h_k},\ b_k=\frac{\sigma_kx_{k+1}-\sigma_{k+1}x_k}{2h_k},\ c_k=\frac{-\sigma_kx_{k+1}^2+\sigma_{k+1}x_k^2}{2h_k}+s_k-\frac{(\sigma_{k+1}-\sigma_k)h_k}{6}, \]
\[ d_k=\frac{\sigma_kx_{k+1}^3-\sigma_{k+1}x_k^3}{6h_k}+\frac{y_kx_{k+1}-y_{k+1}x_k}{h_k}+\frac{(\sigma_{k+1}x_k-\sigma_kx_{k+1})h_k}{6}. \]

\pp{Spline cấp 4.} Đặt $m_k=S'(x_k)$, $\sigma_k=S''(x_k)$, $M_k=S'''(x_k)$ ($S\in C^3$ nên các giá trị này chung cho hai mảnh kề). $S_k'''$ bậc nhất: $S_k'''(x)=M_k+\frac{M_{k+1}-M_k}{h_k}u$, $u=x-x_k$. Tích phân ba lần với giá trị ban đầu tại $x_k$ (khai triển Taylor):
\[ S_k(x)=y_k+m_ku+\frac{\sigma_k}{2}u^2+\frac{M_k}{6}u^3+\frac{M_{k+1}-M_k}{24h_k}u^4,\qquad u=x-x_k. \]
Liên tục của $S,S',S''$ tại $x_{k+1}$ ($S'''$ liên tục sẵn), $k=0..n-1$:
\[ \boxed{\begin{aligned}
&y_k+m_kh_k+\frac{\sigma_k}{2}h_k^2+\frac{3M_k+M_{k+1}}{24}h_k^3=y_{k+1},\\
&m_k+\sigma_kh_k+\frac{2M_k+M_{k+1}}{6}h_k^2=m_{k+1},\\
&\sigma_k+\frac{M_k+M_{k+1}}{2}h_k=\sigma_{k+1}.
\end{aligned}} \]
$3n$ phương trình, $3(n+1)$ ẩn $(m_k,\sigma_k,M_k)$ $\Rightarrow$ cần 3 điều kiện biên (vd $S'(x_0),S''(x_0),S'(x_n)$). Dạng chính tắc: khai triển $u=x-x_k$ bằng Horner nhân.

\subsection{b) Thuật toán spline cấp $p\le4$}
\begin{thuattoan}{Spline cấp 1}
\item[Input:] vector $x$, $y$.
\item[Output:] $S_k(x)=a_kx+b_k$, $k=0..n-1$.
\item[Bước 0:] \textbf{Xử lý dữ liệu:} Nếu $x$, $y$ khác số phần tử thì báo lỗi, dừng thuật toán.\newline Sắp các mốc tăng theo $x$.\newline Nếu hai mốc trùng nhau thì báo lỗi, dừng thuật toán.\newline Đặt $n:=(\text{số mốc})-1$.\newline Nếu $n<1$ thì báo lỗi (cần ít nhất 2 mốc), dừng thuật toán.
\item[Bước 1:] \textbf{Hệ số:} Lặp $k=0,\dots,n-1$:\thut tính $h_k:=x_{k+1}-x_k$;\thut tính $a_k:=\dfrac{y_{k+1}-y_k}{h_k}$, $b_k:=\dfrac{y_kx_{k+1}-y_{k+1}x_k}{h_k}$.\newline Dừng thuật toán, trả về các $S_k$.
\end{thuattoan}

\begin{thuattoan}{Spline cấp 2}
\item[Input:] vector $x$, $y$; một điều kiện biên $S'(x_0)=A$ hoặc $S'(x_n)=B$.
\item[Output:] $S_k(x)=a_kx^2+b_kx+c_k$, $k=0..n-1$.
\item[Bước 0:] \textbf{Xử lý dữ liệu:} Nếu $x$, $y$ khác số phần tử thì báo lỗi, dừng thuật toán.\newline Sắp các mốc tăng theo $x$.\newline Nếu hai mốc trùng nhau thì báo lỗi, dừng thuật toán.\newline Đặt $n:=(\text{số mốc})-1$.\newline Nếu $n<1$ thì báo lỗi (cần ít nhất 2 mốc), dừng thuật toán.\newline Nếu số điều kiện biên khác $1$ hoặc có điều kiện không đặt tại $x_0$, $x_n$ thì báo lỗi, dừng thuật toán.
\item[Bước 1:] \textbf{Bước lưới:} Lặp $k=0,\dots,n-1$: tính $h_k:=x_{k+1}-x_k$ và $\gamma_k:=\dfrac{2(y_{k+1}-y_k)}{h_k}$.
\item[Bước 2:] \textbf{Đạo hàm tại mốc $m_k=S'(x_k)$:} Nếu cho $S'(x_0)=A$ thì:\thut đặt $m_0:=A$;\thut lặp $k=0,\dots,n-1$: tính $m_{k+1}:=\gamma_k-m_k$.\newline Nếu cho $S'(x_n)=B$ thì:\thut đặt $m_n:=B$;\thut lặp $k=n-1,\dots,0$: tính $m_k:=\gamma_k-m_{k+1}$.
\item[Bước 3:] \textbf{Hệ số:} Lặp $k=0,\dots,n-1$: tính
  \[ a_k:=\frac{m_{k+1}-m_k}{2h_k},\ \ b_k:=\frac{m_kx_{k+1}-m_{k+1}x_k}{h_k},\ \ c_k:=\frac{-m_kx_{k+1}^2+m_{k+1}x_k^2}{2h_k}+y_k+\frac{m_kh_k}{2}. \]
  Dừng thuật toán, trả về các $S_k$.
\end{thuattoan}

\begin{thuattoan}{Spline cấp 3}
\item[Input:] vector $x$, $y$; hai điều kiện biên dạng $S'(\xi)=v$ hoặc $S''(\xi)=v$, $\xi\in\{x_0,x_n\}$.
\item[Output:] $S_k(x)=a_kx^3+b_kx^2+c_kx+d_k$, $k=0..n-1$.
\item[Bước 0:] \textbf{Xử lý dữ liệu:} Nếu $x$, $y$ khác số phần tử thì báo lỗi, dừng thuật toán.\newline Sắp các mốc tăng theo $x$.\newline Nếu hai mốc trùng nhau thì báo lỗi, dừng thuật toán.\newline Đặt $n:=(\text{số mốc})-1$.\newline Nếu $n<1$ thì báo lỗi (cần ít nhất 2 mốc), dừng thuật toán.\newline Nếu số điều kiện biên khác $2$ hoặc có điều kiện không đặt tại $x_0$, $x_n$ thì báo lỗi, dừng thuật toán.
\item[Bước 1:] \textbf{Bước lưới:} Lặp $k=0,\dots,n-1$: tính $h_k:=x_{k+1}-x_k$ và $s_k:=\dfrac{y_{k+1}-y_k}{h_k}$.
\item[Bước 2:] \textbf{Lập hệ:} Lập hệ $n+1$ phương trình ẩn $\sigma_0,\dots,\sigma_n$:\thut $n-1$ phương trình trơn (khung, câu a);\thut 2 phương trình biên (câu a).
\item[Bước 3:] \textbf{Truy đuổi:} Nếu mỗi đầu có một điều kiện thì hệ có dạng ba đường chéo $\ell_i\sigma_{i-1}+d_i\sigma_i+u_i\sigma_{i+1}=r_i$, giải bằng truy đuổi:
  \begin{con}
  \item[3.1:] Đặt $P_0:=-u_0/d_0$, $Q_0:=r_0/d_0$.\newline Lặp $i=1,\dots,n$:\thut tính $\mu:=d_i+\ell_iP_{i-1}$; nếu $\mu=0$ thì báo hệ suy biến, dừng thuật toán;\thut tính $P_i:=-u_i/\mu$, $Q_i:=(r_i-\ell_iQ_{i-1})/\mu$.
  \item[3.2:] Đặt $\sigma_n:=Q_n$.\newline Lặp $i=n-1,\dots,0$: tính $\sigma_i:=P_i\sigma_{i+1}+Q_i$.\newline Sang bước 5.
  \end{con}
\item[Bước 4:] \textbf{Khử Gauss:} Nếu hai điều kiện cùng một đầu thì giải hệ bằng khử Gauss chọn trụ; nếu trụ bằng $0$ thì báo ``điều kiện biên không xác định duy nhất spline'', dừng thuật toán.
\item[Bước 5:] \textbf{Hệ số:} Lặp $k=0,\dots,n-1$: tính $a_k,b_k,c_k,d_k$ theo câu a.\newline Dừng thuật toán, trả về các $S_k$.
\end{thuattoan}

\section{Câu 22. Thuật toán phương pháp bình phương tối thiểu}

\subsection{a) Hàm thực nghiệm với tập cơ sở $\{\varphi_j\}_{j=1}^m$ và sai số trung bình phương}
\begin{thuattoan}{Bình phương tối thiểu tuyến tính theo tham số}
\item[Input:] vector $x$, $y$; cơ sở $\varphi_1,\dots,\varphi_m$.
\item[Output:] vector hệ số $a=(a_1,\dots,a_m)$ của $g=\sum_ja_j\varphi_j$; sai số trung bình phương $\sigma$.
\item[Bước 0:] \textbf{Xử lý dữ liệu:} Nếu $x$, $y$ khác số phần tử thì báo lỗi, dừng thuật toán.\newline Loại các điểm làm $\varphi_j(x_i)$ không xác định.\newline Đặt $n:=(\text{số điểm còn lại})$, đánh số lại $i=1,\dots,n$.\newline Nếu $n<m$ thì báo lỗi, dừng thuật toán.
\item[Bước 1:] \textbf{Ma trận cơ sở:} Khởi tạo ma trận $\Phi$ cỡ $n\times m$.\newline Lặp lồng $i=1,\dots,n$; $j=1,\dots,m$: tính $\Phi_{ij}:=\varphi_j(x_i)$.
\item[Bước 2:] \textbf{Hệ chuẩn:} Khởi tạo ma trận $M$ cỡ $m\times m$ và vector $d$ có $m$ phần tử.\newline Lặp lồng $j=1,\dots,m$; $k=1,\dots,m$: tính $M_{jk}:=\sum_{i=1}^n\Phi_{ij}\Phi_{ik}$.\newline Lặp $j=1,\dots,m$: tính $d_j:=\sum_{i=1}^n\Phi_{ij}y_i$.
\item[Bước 3:] \textbf{Giải hệ:} Giải hệ chuẩn $Ma=d$ bằng phân tích Cholesky $M=LL^{T}$ ($M$ đối xứng, xác định dương; $L$ tam giác dưới): giải $Lu=d$ (thế xuôi) rồi $L^{T}a=u$ (thế ngược).\newline Nếu không phân tích được ($M$ không xác định dương) thì báo ``các hàm cơ sở phụ thuộc tuyến tính trên dữ liệu'', dừng thuật toán.
\item[Bước 4:] \textbf{Sai số:} Tính $\sigma:=\sqrt{\frac1n\sum_{i=1}^n\big(\sum_ja_j\Phi_{ij}-y_i\big)^2}$.\newline Dừng thuật toán, trả về $a$ và $\sigma$.
\end{thuattoan}

\subsection{b) Dạng $y=a\,e^{b_1\varphi_1(x)+b_2\varphi_2(x)}$ và $y=a\,x^b$}
\begin{thuattoan}{Dạng mũ $y=a\,e^{b_1\varphi_1(x)+b_2\varphi_2(x)}$}
\item[Input:] vector $x$, $y$; cơ sở $\varphi_1,\varphi_2$ (không phải hàm hằng).
\item[Output:] $a$, $b_1$, $b_2$, lượng tịnh tiến $S_y$ và sai số trung bình phương $\sigma$.
\item[Bước 0:] \textbf{Xử lý dữ liệu:} Nếu $x$, $y$ khác số phần tử thì báo lỗi, dừng thuật toán.\newline Loại các điểm làm $\varphi_1(x_i)$, $\varphi_2(x_i)$ không xác định.\newline Đặt $n:=(\text{số điểm còn lại})$, đánh số lại $i=1,\dots,n$.\newline Đặt $s:=1$ nếu số điểm $y_i>0$ không ít hơn số điểm $y_i<0$, ngược lại $s:=-1$ ($s$ là dấu của $a$).\newline Đặt $q:=(\text{số điểm có }s\,y_i\le0)$, $S_y:=0$.\newline Nếu $0<q\le0.1\,n$ thì loại $q$ điểm đó, đặt lại $n:=n-q$, đánh số lại $i=1,\dots,n$.\newline Nếu $q>0.1\,n$ thì đặt $S_y:=s\,\big(\lfloor\max_{s\,y_i\le0}|y_i|\rfloor+1\big)$ (mọi $s(y_i+S_y)>0$).\newline Nếu $n<3$ thì báo lỗi, dừng thuật toán.
\item[Bước 1:] \textbf{Đổi biến:} Lặp $i=1,\dots,n$: tính $z_i:=\ln\big(s(y_i+S_y)\big)$.
\item[Bước 2:] \textbf{Bình phương tối thiểu:} Tìm $(c_0,c_1,c_2)$ theo câu a với dữ liệu $(x_i,z_i)$ và cơ sở $(1,\varphi_1,\varphi_2)$.
\item[Bước 3:] \textbf{Trả biến, sai số:} Đặt $a:=s\,e^{c_0}$, $b_1:=c_1$, $b_2:=c_2$.\newline Tính $\sigma:=\sqrt{\frac1n\sum_i\big(a\,e^{b_1\varphi_1(x_i)+b_2\varphi_2(x_i)}-S_y-y_i\big)^2}$.\newline Dừng thuật toán, trả về $a$, $b_1$, $b_2$, $S_y$, $\sigma$.
\end{thuattoan}

\begin{thuattoan}{Dạng luỹ thừa $y=a\,x^b$}
\item[Input:] vector $x$, $y$.
\item[Output:] $a$, $b$, lượng tịnh tiến $S_x$, $S_y$ và sai số trung bình phương $\sigma$.
\item[Bước 0:] \textbf{Xử lý dữ liệu:} Nếu $x$, $y$ khác số phần tử thì báo lỗi, dừng thuật toán.\newline Đặt $n:=(\text{số điểm})$.\newline Đặt $q_x:=(\text{số điểm có }x_i\le0)$, $S_x:=0$.\newline Nếu $0<q_x\le0.1\,n$ thì loại $q_x$ điểm đó, đặt lại $n:=n-q_x$, đánh số lại $i=1,\dots,n$.\newline Nếu $q_x>0.1\,n$ thì đặt $S_x:=\lfloor|\min x|\rfloor+1$.\newline Đặt $s:=1$ nếu số điểm $y_i>0$ không ít hơn số điểm $y_i<0$, ngược lại $s:=-1$.\newline Đặt $q:=(\text{số điểm có }s\,y_i\le0)$, $S_y:=0$.\newline Nếu $0<q\le0.1\,n$ thì loại $q$ điểm đó, đặt lại $n:=n-q$, đánh số lại $i=1,\dots,n$.\newline Nếu $q>0.1\,n$ thì đặt $S_y:=s\,\big(\lfloor\max_{s\,y_i\le0}|y_i|\rfloor+1\big)$.\newline Nếu $n<2$ thì báo lỗi, dừng thuật toán.
\item[Bước 1:] \textbf{Đổi biến:} Lặp $i=1,\dots,n$: tính $X_i:=\ln(x_i+S_x)$ và $z_i:=\ln\big(s(y_i+S_y)\big)$.
\item[Bước 2:] \textbf{Bình phương tối thiểu:} Tìm $(c_0,c_1)$ theo câu a với dữ liệu $(X_i,z_i)$ và cơ sở $(1,X)$.
\item[Bước 3:] \textbf{Trả biến, sai số:} Đặt $a:=s\,e^{c_0}$, $b:=c_1$.\newline Tính $\sigma:=\sqrt{\frac1n\sum_i\big(a(x_i+S_x)^b-S_y-y_i\big)^2}$.\newline Dừng thuật toán, trả về $a$, $b$, $S_x$, $S_y$, $\sigma$.
\end{thuattoan}
